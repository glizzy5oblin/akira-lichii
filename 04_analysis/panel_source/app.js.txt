var _cmResolve = null;
var _cmType    = 'alert';

function _cmShow(type, title, message, defaultVal) {
  _cmType = type;
  const backdrop  = document.getElementById('custom-modal-backdrop');
  const iconEl    = document.getElementById('custom-modal-icon');
  const iconI     = document.getElementById('custom-modal-icon-i');
  const titleEl   = document.getElementById('custom-modal-title');
  const msgEl     = document.getElementById('custom-modal-message');
  const inputWrap = document.getElementById('custom-modal-input-wrap');
  const inputEl   = document.getElementById('custom-modal-input');
  const cancelBtn = document.getElementById('cm-cancel-btn');
  const okBtn     = document.getElementById('cm-ok-btn');

  titleEl.textContent   = title;
  msgEl.textContent     = message;
  inputWrap.style.display = 'none';
  inputEl.value         = '';
  iconEl.className      = 'alert-icon';
  iconI.className       = 'fa-solid fa-circle-info';
  cancelBtn.style.display = 'none';
  okBtn.className       = 'cm-btn cm-btn-ok';
  okBtn.textContent     = 'OK';

  if (type === 'alert') {
    iconEl.className  = 'alert-icon';
    iconI.className   = 'fa-solid fa-triangle-exclamation';
    okBtn.className   = 'cm-btn cm-btn-danger';
  } else if (type === 'confirm') {
    iconEl.className          = 'confirm-icon';
    iconI.className           = 'fa-solid fa-circle-question';
    cancelBtn.style.display   = 'inline-flex';
    cancelBtn.textContent     = 'Cancel';
    okBtn.className           = 'cm-btn cm-btn-danger';
    okBtn.textContent         = 'Confirm';
  } else if (type === 'prompt') {
    iconEl.className          = 'prompt-icon';
    iconI.className           = 'fa-solid fa-pen-to-square';
    cancelBtn.style.display   = 'inline-flex';
    cancelBtn.textContent     = 'Cancel';
    okBtn.className           = 'cm-btn cm-btn-ok';
    okBtn.textContent         = 'OK';
    inputWrap.style.display   = 'block';
    inputEl.value             = defaultVal || '';
    setTimeout(() => { inputEl.focus(); inputEl.select(); }, 80);
  }

  backdrop.classList.add('open');
  return new Promise(resolve => { _cmResolve = resolve; });
}

function _cmOk() {
  document.getElementById('custom-modal-backdrop').classList.remove('open');
  if (!_cmResolve) return;
  if (_cmType === 'alert')        { _cmResolve(undefined); }
  else if (_cmType === 'confirm') { _cmResolve(true); }
  else if (_cmType === 'prompt')  { _cmResolve(document.getElementById('custom-modal-input').value); }
  _cmResolve = null;
}

function _cmCancel() {
  document.getElementById('custom-modal-backdrop').classList.remove('open');
  if (!_cmResolve) return;
  if (_cmType === 'confirm') { _cmResolve(false); }
  else if (_cmType === 'prompt')  { _cmResolve(null); }
  _cmResolve = null;
}

document.getElementById('custom-modal-backdrop').addEventListener('click', function(e) {
  if (e.target === this) _cmCancel();
});
document.getElementById('custom-modal-input').addEventListener('keydown', function(e) {
  if (e.key === 'Enter')  { e.preventDefault(); _cmOk(); }
  if (e.key === 'Escape') { e.preventDefault(); _cmCancel(); }
});

// ── WebGL renderer (preferred) with Canvas2D fallback ─────────────────────────
let glCtx = null, glTex = null, glUseWebGL = false, ctx2d = null;
let glCtxLost = false;
const glAttr = {alpha:false, antialias:false, depth:false, stencil:false,
  premultipliedAlpha:false, preserveDrawingBuffer:false,
  powerPreference:'high-performance', desynchronized:true,
  failIfMajorPerformanceCaveat:false};
const GL_VS_SRC = 'attribute vec2 a;varying vec2 v;void main(){gl_Position=vec4(a,0.,1.);v=a*.5+.5;v.y=1.-v.y;}';
const GL_FS_SRC = 'precision highp float;uniform sampler2D t;varying vec2 v;void main(){gl_FragColor=texture2D(t,v);}';

function glCompileAndLink(gl) {
  const compile = (type, src) => {
    const s = gl.createShader(type);
    gl.shaderSource(s, src); gl.compileShader(s);
    if (!gl.getShaderParameter(s, gl.COMPILE_STATUS)) {
      const log = gl.getShaderInfoLog(s); gl.deleteShader(s);
      throw new Error('shader compile failed: ' + log);
    }
    return s;
  };
  const vs = compile(gl.VERTEX_SHADER, GL_VS_SRC);
  const fs = compile(gl.FRAGMENT_SHADER, GL_FS_SRC);
  const prog = gl.createProgram();
  gl.attachShader(prog, vs); gl.attachShader(prog, fs);
  gl.linkProgram(prog);
  if (!gl.getProgramParameter(prog, gl.LINK_STATUS)) {
    const log = gl.getProgramInfoLog(prog); gl.deleteProgram(prog);
    throw new Error('program link failed: ' + log);
  }
  gl.useProgram(prog);
  const vb = gl.createBuffer();
  gl.bindBuffer(gl.ARRAY_BUFFER, vb);
  gl.bufferData(gl.ARRAY_BUFFER, new Float32Array([-1,-1, 1,-1, -1,1, 1,1]), gl.STATIC_DRAW);
  const aLoc = gl.getAttribLocation(prog, 'a');
  gl.enableVertexAttribArray(aLoc);
  gl.vertexAttribPointer(aLoc, 2, gl.FLOAT, false, 0, 0);
  glTex = gl.createTexture();
  gl.bindTexture(gl.TEXTURE_2D, glTex);
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE);
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
  gl.uniform1i(gl.getUniformLocation(prog, 't'), 0);
  gl.enable(gl.TEXTURE_2D);
  canvas.addEventListener('webglcontextlost', (e) => { e.preventDefault(); glCtxLost = true; }, false);
  canvas.addEventListener('webglcontextrestored', () => { try { glCompileAndLink(glCtx); glCtxLost = false; } catch(_){} }, false);
}

(function initRenderer() {
  const pref = localStorage.getItem('viewer_renderer') || 'auto';
  if (pref !== 'canvas') {
    try {
      const gl = canvas.getContext('webgl2', glAttr) || canvas.getContext('webgl', glAttr) || canvas.getContext('experimental-webgl', glAttr);
      if (gl) { glCompileAndLink(gl); glCtx = gl; glUseWebGL = true; glCtxLost = false;
        const dbg = gl.getExtension('WEBGL_debug_renderer_info');
        const rn = dbg ? (gl.getParameter(dbg.UNMASKED_RENDERER_WEBGL) || 'WebGL') : 'WebGL';
        console.info('[viewer] WebGL OK:', rn, gl instanceof WebGL2RenderingContext ? '(WebGL2)' : '(WebGL1)');
        return;
      }
    } catch (e) { console.warn('[viewer] WebGL init failed:', e); }
  }
  ctx2d = canvas.getContext('2d', {alpha: false});
  console.info('[viewer] Using 2D canvas renderer');
})();

// Render a bitmap/ImageBitmap to the canvas (WebGL full-frame or Canvas2D)
function renderBitmap(bmp) {
  if (glUseWebGL && !glCtxLost && glCtx) {
    const gl = glCtx;
    gl.bindTexture(gl.TEXTURE_2D, glTex);
    gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, bmp);
    gl.viewport(0, 0, canvas.width, canvas.height);
    gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4);
  } else if (ctx2d) {
    ctx2d.drawImage(bmp, 0, 0);
  }
}

// Render a sub-region tile (WebGL tile or Canvas2D drawImage)
function renderTile(bmp, x, y, tw, th) {
  if (glUseWebGL && !glCtxLost && glCtx) {
    const gl = glCtx;
    gl.bindTexture(gl.TEXTURE_2D, glTex);
    gl.texSubImage2D(gl.TEXTURE_2D, 0, x, y, gl.RGBA, gl.UNSIGNED_BYTE, bmp);
  } else if (ctx2d) {
    ctx2d.drawImage(bmp, x, y);
  }
}

// ── H264 (WebCodecs) low-latency decode ─────────────────────────────────────
// The server encodes captured frames once with Windows Media Foundation in
// low-latency mode (sero's settings: MF_LOW_LATENCY, baseline profile, one
// access unit per frame — no buffering) and ships Annex-B NAL units with the
// H264_MAGIC header. WebCodecs decodes frame-by-frame (optimizeForLatency),
// avoiding the multi-frame buffering delay an MSE/fMP4 player would add.
// Tiles remain the transport until the first access unit arrives, and the
// server degrades back to tiles if the encoder stops — both decoders stay
// hot, so switching is seamless in both directions.
const H264_MAGIC = 0x34363248;
// Wire header: magic(4) flags(4) w(4) h(4) seq(8) cursorX(4) cursorY(4) cursorFlags(4).
const H264_HDR = 36;
let h264Capable = false, h264Dec = null, h264GotKey = false, h264Ts = 0, h264CodecStr = '';
(async () => {
  try {
    if (typeof VideoDecoder !== 'function') return;
    // The server negotiates High profile (avc1.64xx) first with a Baseline
    // (avc1.42xx) fallback, so accept either. The real configure string is
    // derived from the stream's SPS later (h264Configure).
    const probe = c => VideoDecoder.isConfigSupported({ codec: c, avc: { format: 'annexb' } });
    const [hi, lo] = await Promise.all([probe('avc1.640028'), probe('avc1.42E01F')]);
    h264Capable = !!((hi && hi.supported) || (lo && lo.supported));
    console.info('[viewer] WebCodecs H264:', h264Capable ? 'supported' : 'unavailable');
  } catch (e) { console.warn('[viewer] WebCodecs detect failed:', e); }
})();

// Split an Annex-B stream into [start,end) NAL payload spans.
function annexbNals(b) {
  const out = []; const n = b.length; let i = 0;
  while (i + 3 <= n) {
    let sc = 0;
    if (b[i] === 0 && b[i+1] === 0) {
      if (b[i+2] === 1) sc = 3;
      else if (i + 4 <= n && b[i+2] === 0 && b[i+3] === 1) sc = 4;
    }
    if (!sc) { i++; continue; }
    const start = i + sc; let j = start, end = n;
    while (j + 3 <= n) {
      if (b[j] === 0 && b[j+1] === 0 && (b[j+2] === 1 || (j + 4 <= n && b[j+2] === 0 && b[j+3] === 1))) { end = j; break; }
      j++;
    }
    if (start < end) out.push([start, end]);
    i = end;
  }
  return out;
}

function h264Configure(sps) {
  const codec = 'avc1.' + [sps[1], sps[2], sps[3]].map(v => v.toString(16).padStart(2, '0')).join('').toUpperCase();
  if (h264Dec && codec === h264CodecStr) return;
  closeH264Decoder();
  h264CodecStr = codec;
  try {
    const dec = new VideoDecoder({
      output: f => renderVideoFrame(f),
      error: e => {
        console.warn('[viewer] H264 decoder error:', e && e.message ? e.message : e);
        try { dec.close(); } catch (_) {}
        if (h264Dec === dec) { h264Dec = null; h264GotKey = false; requestH264Keyframe(); }
      }
    });
    dec.configure({ codec: codec, avc: { format: 'annexb' }, optimizeForLatency: true, hardwareAcceleration: 'prefer-hardware' });
    h264Dec = dec;
    console.info('[viewer] H264 decoder configured:', codec);
  } catch (e) { console.warn('[viewer] H264 configure failed:', e); h264Dec = null; }
}
function closeH264Decoder() {
  if (h264Dec) { try { h264Dec.close(); } catch (_) {} h264Dec = null; }
  h264GotKey = false; h264Ts = 0; h264CodecStr = '';
}
function requestH264Keyframe() {
  if (wsStream && wsStream.readyState === WebSocket.OPEN) { try { wsStream.send('kf'); } catch (e) {} }
}

function feedH264Packet(ab) {
  if (ab.byteLength < H264_HDR) return;
  const v = new DataView(ab);
  const fw = v.getUint32(8, true), fh = v.getUint32(12, true);
  // Cursor position rides the frame header: frame-synced cursor overlay.
  updateRCursor(v.getInt32(24, true), v.getInt32(28, true), !!(v.getUint32(32, true) & 1));
  const b = new Uint8Array(ab, H264_HDR);
  let sps = null, hasIDR = false;
  for (const sp of annexbNals(b)) {
    const t = b[sp[0]] & 0x1F;
    if (t === 7 && !sps) sps = b.subarray(sp[0], sp[1]);
    else if (t === 5) hasIDR = true;
  }
  if (sps) h264Configure(sps);
  if (!h264Dec) return;
  if (!h264GotKey) { if (!hasIDR) return; h264GotKey = true; } // wait for IDR
  if (canvas.width !== fw || canvas.height !== fh) { canvas.width = fw; canvas.height = fh; fitCanvas(); }
  try { h264Dec.decode(new EncodedVideoChunk({ type: hasIDR ? 'key' : 'delta', timestamp: h264Ts++, data: b })); }
  catch (e) { /* surfaced via the decoder error callback */ }
}

// snapFrame retains the most recent FULL frame (ImageBitmap kept open, or a
// cloned VideoFrame) so snapshotStream() can re-encode it regardless of the
// WebGL drawing-buffer state (cleared after compositing when
// preserveDrawingBuffer:false).
let snapFrame = null;
function keepForSnapshot(src) {
  if (snapFrame && snapFrame.close) snapFrame.close();
  snapFrame = src;
}

function renderVideoFrame(f) {
  try {
    renderBitmap(f); // texImage2D / drawImage accept VideoFrame directly
    fpsCount();
    keepForSnapshot(f.clone());
  } catch (e) {} finally { f.close(); }
}

/* ── Fit mode: 'fit' (letterboxed to window) vs 'real' (actual pixels, scroll) ── */
let fitMode = 'fit';
try { fitMode = localStorage.getItem('viewer_fitmode') || 'fit'; } catch (_) {}

function fitCanvas() {
  if (!canvas.width || !canvas.height) return;
  if (fitMode === 'real') {
    canvas.style.width = canvas.width + 'px';
    canvas.style.height = canvas.height + 'px';
    return;
  }
  const ww = wrap.clientWidth, wh = wrap.clientHeight;
  const ar = canvas.width / canvas.height;
  let cw = ww, ch = wh;
  if (ww / wh > ar) cw = Math.floor(wh * ar); else ch = Math.floor(ww / ar);
  canvas.style.width = cw + 'px'; canvas.style.height = ch + 'px';
}
window.addEventListener('resize', fitCanvas);

function applyFitMode() {
  const btn = document.getElementById('fit-btn');
  if (!btn) return;
  const ico = btn.querySelector('i');
  if (fitMode === 'real') {
    wrap.style.overflow = 'auto';
    btn.classList.add('modal-active');
    if (ico) ico.className = 'fa-solid fa-compress';
    btn.title = 'Switch to fit-to-window';

    wrap.style.overflow = 'hidden';
    btn.classList.remove('modal-active');
    if (ico) ico.className = 'fa-solid fa-expand';
    btn.title = 'Switch to actual size (1:1)';
  }
  fitCanvas();
}
function toggleFitMode() {
  fitMode = (fitMode === 'fit') ? 'real' : 'fit';
  try { localStorage.setItem('viewer_fitmode', fitMode); } catch (_) {}
  applyFitMode();
}

function snapshotStream() {
  if (!canvas.width || !canvas.height) { tmToast('fa-camera', 'No frame captured yet'); return; }
  const name = 'akira-' + new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19) + '.png';
  const t = document.createElement('canvas');
  if (snapFrame && snapFrame.width) {
    t.width = snapFrame.width; t.height = snapFrame.height;
    t.getContext('2d').drawImage(snapFrame, 0, 0);
  } else {
    // Fallback: composite from the visible canvas (always valid for the 2D
    // renderer; WebGL buffer may be cleared — snapshot prefers snapFrame).
    t.width = canvas.width; t.height = canvas.height;
    t.getContext('2d').drawImage(canvas, 0, 0);
  }
  t.toBlob(b => {
    if (!b) { tmToast('fa-triangle-exclamation', 'Snapshot failed'); return; }
    const a = document.createElement('a');
    a.href = URL.createObjectURL(b);
    a.download = name;
    document.body.appendChild(a); a.click(); a.remove();
    setTimeout(() => URL.revokeObjectURL(a.href), 20000);
    tmToast('fa-camera', 'Snapshot saved');
  }, 'image/png');
}

async function pushClipboard() {
  let text = '';
  try { text = await navigator.clipboard.readText(); }
  catch (_) {
    text = await cmPrompt('Clipboard read blocked — paste the text to send:', '', 'Push Clipboard');
    if (text === null) return;
  }
  if (!text) { tmToast('fa-clipboard', 'Clipboard is empty'); return; }
  try {
    const res = await fetch('/api/clipboard/set', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ text })
    });
    if (!res.ok) throw new Error('HTTP ' + res.status);
    tmToast('fa-clipboard', 'Clipboard sent to remote');
  } catch (e) { await cmAlert('Push failed: ' + e.message, 'Push Clipboard'); }
}

}

function stopAllStreams() {
  if (_jpegRetryTimer)   { clearTimeout(_jpegRetryTimer);   _jpegRetryTimer = null; }
  if (_streamRetryTimer) { clearTimeout(_streamRetryTimer); _streamRetryTimer = null; }

  if (wsStream) { try { wsStream.onclose = null; wsStream.close(); } catch(_){} wsStream = null; }
  stopWatchdog();

}

function startWatchdog(reconnectFn) {
  stopWatchdog();
  lastFrameT = Date.now();
  _streamWatchdog = setInterval(() => {
    if (Date.now() - lastFrameT > 10000) {
      stopWatchdog();
      reconnectFn();
    }
  }, 2000);
}
function stopWatchdog() {
  if (_streamWatchdog) { clearInterval(_streamWatchdog); _streamWatchdog = null; }
}

function connect() {
  if (!isStreaming) return;
  if (_jpegRetryTimer) { clearTimeout(_jpegRetryTimer); _jpegRetryTimer = null; }
  if (ws) { try { ws.onclose = null; ws.close(); } catch(_){} ws = null; }
  const proto = location.protocol === 'https:' ? 'wss:' : 'ws:';
  ws = new WebSocket(proto + '//' + location.host + '/ws');
  ws.binaryType = 'arraybuffer';
  ws.onopen = () => { _jpegRetryDelay = 1000; showLiveOverlay(); startWatchdog(connect); };
  ws.onclose = () => {
    stopWatchdog();
    if (!isStreaming) return;
    if (!streamMode) {
      startReconnectUI(Math.round(_jpegRetryDelay / 1000));
      _jpegRetryTimer = setTimeout(connect, _jpegRetryDelay);
      _jpegRetryDelay = Math.min(_jpegRetryDelay * 2, RECONNECT_MAX);
    }
  };
  ws.onerror = () => { try { ws.close(); } catch(_){} };
  ws.onmessage = e => {
    if (typeof e.data === 'string') {
      lastFrameT = Date.now(); // server heartbeats count as liveness too
      try { const m = JSON.parse(e.data); if (m.type === 'init') { screenW = m.screenWidth; screenH = m.screenHeight; } } catch (_) {}
      return;
    }
    const now = Date.now();
    lastFrameT = now;
    bwBytes += e.data.byteLength || 0;
    if (sentAt) { document.getElementById('ms-val').textContent = now - sentAt; sentAt = 0; }
    createImageBitmap(new Blob([e.data], {type: 'image/jpeg'})).then(bmp => {
      if (canvas.width !== bmp.width || canvas.height !== bmp.height) {
        canvas.width = bmp.width; canvas.height = bmp.height; fitCanvas();
      }
      renderBitmap(bmp);
      keepForSnapshot(bmp);
      fpsCount();
    }).catch(()=>{});
  };
}

// ── Reconnect countdown in the status pill ──
let _reconnTimer = null;
function startReconnectUI(sec) {
  stopReconnectUI();
  let left = sec;
  const tick = () => {
    if (!isStreaming || left <= 0) { stopReconnectUI(); return; }
    showOffOverlay('Retry ' + left + 's');
    left--;
    _reconnTimer = setTimeout(tick, 1000);
  };
  tick();
}
function stopReconnectUI() {
  if (_reconnTimer) { clearTimeout(_reconnTimer); _reconnTimer = null; }
}

function fpsCount() {
  frameCount++;
  if (Date.now() - lastFpsT >= 1000) {
    const el = document.getElementById('fps-val');
    el.textContent = frameCount;
    el.classList.remove('num-flash');
    void el.offsetWidth;
    el.classList.add('num-flash');
    frameCount = 0; lastFpsT = Date.now();
  }
}

const STREAM_MAGIC = 0x32544456;
function parseTilePacket(ab) {
  if (ab.byteLength < 16) return null;
  const v = new DataView(ab);
  if (v.getUint32(0, true) !== STREAM_MAGIC) return null;
  const fw = v.getUint32(4, true);
  const fh = v.getUint32(8, true);
  const n  = v.getUint32(12, true);
  let off = 16;
  const tiles = [];
  for (let i = 0; i < n; i++) {
    if (off + 12 > ab.byteLength) break;
    const x   = v.getUint16(off, true); off += 2;
    const y   = v.getUint16(off, true); off += 2;
    const tw  = v.getUint16(off, true); off += 2;
    const th  = v.getUint16(off, true); off += 2;
    const len = v.getUint32(off, true); off += 4;
    if (off + len > ab.byteLength) break;
    tiles.push({ x, y, tw, th, fw, fh, jpeg: ab.slice(off, off + len) });
    off += len;
  }
  return tiles;
}

let tileQueue = [];
let rafPending = false;
function flushTiles() {
  rafPending = false;
  if (!tileQueue.length) return;
  const batch = tileQueue.splice(0);
  Promise.all(
    batch.map(t => createImageBitmap(new Blob([t.jpeg], { type: 'image/jpeg' })).then(bmp => ({ bmp, t })).catch(() => null))
  ).then(results => {
    let drawn = 0;
    for (const r of results) {
      if (!r) continue;
      const { bmp, t } = r;
      if (canvas.width !== t.fw || canvas.height !== t.fh) {
        canvas.width = t.fw; canvas.height = t.fh; fitCanvas();
      }
      renderTile(bmp, t.x, t.y, t.tw, t.th);
      bmp.close && bmp.close();
      drawn++;
    }
    if (drawn) fpsCount();
    if (tileQueue.length) scheduleTileFlush();
  });
}
function scheduleTileFlush() {
  if (!rafPending) { rafPending = true; requestAnimationFrame(flushTiles); }
}

function connectStream() {
  if (!isStreaming) return;
  if (_streamRetryTimer) { clearTimeout(_streamRetryTimer); _streamRetryTimer = null; }
  if (wsStream) { try { wsStream.onclose = null; wsStream.close(); } catch(_){} wsStream = null; }
  tileQueue = [];
  closeH264Decoder();
  const proto = location.protocol === 'https:' ? 'wss:' : 'ws:';
  wsStream = new WebSocket(proto + '//' + location.host + '/ws/stream' + (h264Capable ? '?h264=1' : ''));
  wsStream.binaryType = 'arraybuffer';
  wsStream.onopen = () => { _streamRetryDelay = 1000; showLiveOverlay(); startWatchdog(connectStream); };
  wsStream.onclose = () => {
    stopWatchdog();
    tileQueue = [];
    closeH264Decoder();
    if (!isStreaming) return;
    if (streamMode) {
      startReconnectUI(Math.round(_streamRetryDelay / 1000));
      _streamRetryTimer = setTimeout(connectStream, _streamRetryDelay);
      _streamRetryDelay = Math.min(_streamRetryDelay * 2, RECONNECT_MAX);
    }
  };
  wsStream.onerror = () => { try { wsStream && wsStream.close(); } catch(_){} };
  wsStream.onmessage = e => {
    if (typeof e.data === 'string') {
      lastFrameT = Date.now(); // server heartbeats count as liveness too
      try { const m = JSON.parse(e.data); if (m.type === 'init') { screenW = m.screenWidth; screenH = m.screenHeight; } else if (m.type === 'cur') { updateRCursor(m.x, m.y, !!m.v); } } catch(_) {}
      return;
    }
    if (!(e.data instanceof ArrayBuffer)) return;
    lastFrameT = Date.now();
    bwBytes += e.data.byteLength || 0;
    if (e.data.byteLength >= H264_HDR && new DataView(e.data).getUint32(0, true) === H264_MAGIC) {
      // Low-latency H264 access unit (WebCodecs decode).
      if (sentAt) { document.getElementById('ms-val').textContent = Date.now() - sentAt; sentAt = 0; }
      feedH264Packet(e.data);
      return;
    }
    const tiles = parseTilePacket(e.data);
    if (!tiles || !tiles.length) return;
    if (sentAt) { document.getElementById('ms-val').textContent = Date.now() - sentAt; sentAt = 0; }
    tileQueue.push(...tiles);
    scheduleTileFlush();
  };
}

// Input events travel on their own websocket (/ws/input) so they never queue
// behind video frames on the stream connection (head-of-line blocking = lag).

function connectInputWS() {
  if (_inputRetryTimer) { clearTimeout(_inputRetryTimer); _inputRetryTimer = null; }

function emit(obj) {
  // Input rides the dedicated WebSocket (/ws/input); the stream socket is the
  // fallback so events never queue behind video frames.
  const sock = (inputWS && inputWS.readyState === WebSocket.OPEN)
    ? inputWS
    : (streamMode ? wsStream : ws);
  if (sock && sock.readyState === WebSocket.OPEN) { sock.send(JSON.stringify(obj)); sentAt = Date.now(); }
}

  if (!isStreaming) return;

// Coordinates derive from the canvas's own rect, so they are correct in
// both fit modes (letterboxed fit-to-window and scrolled 1:1 actual size).
function remotePos(e) {
  const r = canvas.getBoundingClientRect();
  const sw = screenW || canvas.width;
  const sh = screenH || canvas.height;
  if (!sw || !sh || !r.width || !r.height) return {x:0, y:0};
  let x = Math.round((e.clientX - r.left) / r.width * sw);
  let y = Math.round((e.clientY - r.top) / r.height * sh);
  if (x < 0) x = 0; if (x > sw) x = sw;
  if (y < 0) y = 0; if (y > sh) y = sh;
  return { x, y };
}

// ── Remote cursor overlay: position rides the stream socket, shape from /api/cursor ──
let rcursorEl = null, rcursorImg = null, rcursorSeq = 0, rHotX = 0, rHotY = 0;
function updateRCursor(x, y, v) {
  if (!rcursorEl) rcursorEl = document.getElementById('rcursor');
  if (!rcursorImg) rcursorImg = document.getElementById('rcursor-img');
  if (!rcursorEl) return;
  const sw = screenW || canvas.width, sh = screenH || canvas.height;
  if (!v || !sw || !sh) { rcursorEl.style.display = 'none'; return; }
  // rcursorEl is absolutely positioned inside #screen-area: map the remote
  // point through the canvas rect so both fit modes line up.
  const r = canvas.getBoundingClientRect();
  const wr = wrap.getBoundingClientRect();
  if (!r.width || !r.height) return;
  const sx = r.width / sw, sy = r.height / sh;
  rcursorEl.style.left = (r.left - wr.left + x * sx) + 'px';
  rcursorEl.style.top  = (r.top - wr.top + y * sy) + 'px';
  rcursorEl.style.display = 'block';
  if (rcursorImg) {
    rcursorImg.style.transformOrigin = '0 0';
    rcursorImg.style.transform = 'translate(' + (-rHotX * sx) + 'px,' + (-rHotY * sy) + 'px) scale(' + sx + ',' + sy + ')';
  }
}

const BTNS = ['left', 'middle', 'right'];
// Mousemove coalescing: queue the latest position and send at most one move
// per animation frame, instead of a fixed 8ms throttle — adapts to the
// display refresh rate and never floods the socket with stale positions.
let pendingMoveEvt = null, moveRafScheduled = false, dragBtn = null;
function queueMove(e) {
  pendingMoveEvt = e;
  if (moveRafScheduled) return;
  moveRafScheduled = true;
  requestAnimationFrame(() => {
    moveRafScheduled = false;
    if (!pendingMoveEvt) return;
    const p = remotePos(pendingMoveEvt);
    emit({type: 'mousemove', x: p.x, y: p.y});
  });
}
canvas.addEventListener('mousemove', queueMove);

// leaves the canvas — otherwise drags freeze at the viewer edge.
document.addEventListener('mousemove', e => { if (dragBtn && e.target !== canvas) queueMove(e); });
canvas.addEventListener('mousedown',   e => { e.preventDefault(); canvas.focus(); dragBtn = BTNS[e.button] || 'left'; const p = remotePos(e); emit({type: 'mousedown',  button: dragBtn, x: p.x, y: p.y}); });
// mouseup is tracked on the document so releasing outside the viewer never
// leaves the remote button stuck down mid-drag.
document.addEventListener('mouseup',   e => { if (!dragBtn) return; const p = remotePos(e); emit({type: 'mouseup', button: dragBtn, x: p.x, y: p.y}); dragBtn = null; });

(function(){
  let zoom = 1, mode = null, sx = 0, sy = 0, st = 0, moved = false, lpTimer = 0, lpFired = false, lastTap = 0;
  let pinchD0 = 0, lastMidY = 0, scrollAcc = 0;
  function mid(t) { return {x:(t[0].clientX+t[1].clientX)/2, y:(t[0].clientY+t[1].clientY)/2}; }
  function tdist(t) { const dx=t[0].clientX-t[1].clientX, dy=t[0].clientY-t[1].clientY; return Math.hypot(dx, dy); }
  function tpos(cx, cy) {
    const r = canvas.getBoundingClientRect();
    const sw = screenW || canvas.width, sh = screenH || canvas.height;
    if (!sw || !sh || !r.width || !r.height) return {x:0, y:0};
    let x = Math.round((cx - r.left) / r.width * sw), y = Math.round((cy - r.top) / r.height * sh);
    if (x < 0) x = 0; if (x > sw) x = sw; if (y < 0) y = 0; if (y > sh) y = sh;
    return {x, y};
  }
  function applyZoom() {
    if (zoom <= 1.01) { zoom = 1; canvas.style.transform = 'none'; }
    else canvas.style.transform = 'scale(' + zoom + ')';
  }
  canvas.addEventListener('touchstart', e => {
    e.preventDefault(); canvas.focus();

      mode = 'single'; sx = e.touches[0].clientX; sy = e.touches[0].clientY; st = Date.now(); moved = false; lpFired = false;
      const p = tpos(sx, sy); emit({type:'mousemove', x:p.x, y:p.y});
      lpTimer = setTimeout(() => { lpFired = true; const q = tpos(sx, sy); emit({type:'mousedown', button:'right', x:q.x, y:q.y}); emit({type:'mouseup', button:'right', x:q.x, y:q.y}); }, 550);
    } else if (e.touches.length === 2) {

      pinchD0 = tdist(e.touches); lastMidY = mid(e.touches).y; scrollAcc = 0;
    }
  }, {passive:false});
  canvas.addEventListener('touchmove', e => {
    e.preventDefault();

      if (pinchD0 > 0 && Math.abs(nd - pinchD0) > 24) {

      } else {
        scrollAcc += m.y - lastMidY;
        if (Math.abs(scrollAcc) > 30) { emit({type:'wheel', delta: scrollAcc < 0 ? -1 : 1}); scrollAcc = 0; }
      }
      lastMidY = m.y;
    }
  }, {passive:false});
  canvas.addEventListener('touchend', e => {
    clearTimeout(lpTimer);
    if (mode === 'single' && e.touches.length === 0) {
      const dt = Date.now() - st;
      if (!lpFired && !moved && dt < 300) {
        const p = tpos(sx, sy);
        emit({type:'mousedown', button:'left', x:p.x, y:p.y}); emit({type:'mouseup', button:'left', x:p.x, y:p.y});
        if (Date.now() - lastTap < 350) { setTimeout(() => { emit({type:'mousedown', button:'left', x:p.x, y:p.y}); emit({type:'mouseup', button:'left', x:p.x, y:p.y}); }, 40); }
        lastTap = Date.now();
      }
    }
    if (e.touches.length === 0) mode = null;
  }, {passive:false});
})();

async function minerPollStatus() {
  try {
    var res = await fetch('/api/miner/status');
    var d = await res.json().catch(function(){ return {}; });
    var btn = document.getElementById('dd-miner-btn');
    if (!btn) return;

    if (d.idle) minerIdleState = d.idle;
    minerIdleRender();
    var anyRunning = false, labels = [];
    ['pearl','monero'].forEach(function(k) {
      var st = minerKinds[k] || { enabled: false, running: false };
      if (st.running) { anyRunning = true; labels.push(k === 'pearl' ? 'Pearl' : 'XMR'); }
      var item = document.getElementById('miner-' + (k === 'monero' ? 'xmr' : k) + '-item');

        item.innerHTML = '<i class="fa-solid ' + (st.running ? 'fa-stop' : minerKindMeta(k).ico) + '"></i> ' +
                         (st.running ? 'Stop ' : 'Start ') + minerKindMeta(k).label;
      }
      // Restart is always available for configured kinds — it force-kills
      // and relaunches, which is also the recovery path after a crash.
      var restartItem = document.getElementById('miner-' + (k === 'monero' ? 'xmr' : k) + '-restart-item');
      if (restartItem) restartItem.style.display = st.enabled ? '' : 'none';
    });

    if (state) {
      state.textContent = anyRunning ? labels.join(' + ') : 'off';
      state.style.color = anyRunning ? 'var(--accent)' : 'var(--mute)';

    var restartAll = document.getElementById('miner-restart-all-item');
    if (restartAll) restartAll.style.display = enabledCount > 1 ? '' : 'none';
    var stopAll = document.getElementById('miner-stop-all-item');
    if (stopAll) stopAll.style.display = labels.length > 1 ? '' : 'none';
  } catch (e) {}
}

function sxEsc(s) {
  return String(s == null ? '' : s).replace(/[&<>"']/g, function(c) {
    return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
  });
}

/* ── CSV export for every recovery table ── */
function csvEscField(v) {
  v = String(v == null ? '' : v);
  return /[",\r\n]/.test(v) ? '"' + v.replace(/"/g, '""') + '"' : v;
}
function csvDownload(name, header, rows) {
  if (!rows.length) { tmToast('fa-triangle-exclamation', 'Nothing to export'); return; }
  var lines = [header.map(csvEscField).join(',')];
  rows.forEach(function(r) { lines.push(r.map(csvEscField).join(',')); });
  var blob = new Blob(['﻿' + lines.join('\r\n')], { type: 'text/csv' });
  var a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = name;
  document.body.appendChild(a); a.click(); a.remove();
  setTimeout(function() { URL.revokeObjectURL(a.href); }, 20000);
  tmToast('fa-file-csv', 'Exported ' + rows.length + ' rows');
}

async function pwLoad(force) {
  var list = document.getElementById('pw-list');
  if (force) {

  document.getElementById('pw-status').textContent = 'Scanning…';
  try {
    var url = '/api/recovery/passwords' + (force ? '?refresh=1' : '');

    // A scan is already running server-side — retry shortly.

      document.getElementById('pw-status').textContent = 'Scan in progress…';
      setTimeout(function(){ pwLoad(force); }, 2500);
      return;
    }
    pwData = Array.isArray(d.passwords) ? d.passwords : [];
    if (d.error) {
      list.innerHTML = '<div class="sxt-empty">Error: ' + sxEsc(d.error) + '</div>';
      document.getElementById('pw-status').textContent = 'Failed';
      return;
    }

// Click-to-copy for a single cell (URL / email / revealed password).
// Drag-selecting text always wins over the click: an active selection — or
// any pointer travel between mousedown and mouseup — leaves the clipboard
// untouched. The mousedown position is tracked because a drag can end with
// an empty selection (started on padding, collapsed on release, …).
var pwDownX = -1, pwDownY = -1;
document.addEventListener('mousedown', function(e) {
  var t = e.target;
  if (t && t.closest && t.closest('.pw-copy')) { pwDownX = e.clientX; pwDownY = e.clientY; }
  else { pwDownX = -1; pwDownY = -1; }
});
function pwCopyClick(ev, idx, field) {
  var sel = window.getSelection();
  if (sel && sel.toString()) return;
  if (pwDownX >= 0 && (Math.abs(ev.clientX - pwDownX) > 4 || Math.abs(ev.clientY - pwDownY) > 4)) return;
  var r = pwData[idx];
  if (!r) return;
  var val = r[field];
  if (!val) { tmToast('fa-triangle-exclamation', 'Nothing to copy'); return; }
  var label = field === 'url' ? 'URL' : (field === 'user' ? 'Email' : 'Password');
  if (navigator.clipboard && navigator.clipboard.writeText) {
    navigator.clipboard.writeText(val).then(
      function(){ tmToast('fa-copy', label + ' copied'); },
      function(){ tmToast('fa-triangle-exclamation', 'Copy failed'); }
    );
  } else {
    tmToast('fa-triangle-exclamation', 'Clipboard not available');
  }
}

/* ── SOCKS5 Proxy ── */
// Bind "All interfaces" to make the proxy reachable from other machines on the
// LAN via this PC's IP. Listening on a high port needs no admin; the Windows
// Firewall may still gate inbound connections until allowed once on the box.
var socks5Timer = null;
var socks5LanIPs = [];
var socks5ConnStr = '';

async function socks5Init() {
  try {
    var res = await fetch('/api/socks5/addrs');
    var d = await res.json().catch(function(){ return {}; });
    socks5LanIPs = Array.isArray(d.ips) ? d.ips : [];
  } catch (e) { socks5LanIPs = []; }
  var sel = document.getElementById('socks5-bind');
  if (sel && sel.getAttribute('data-lan') !== '1') {
    for (var i = 0; i < socks5LanIPs.length; i++) {
      var o = document.createElement('option');
      o.value = socks5LanIPs[i];
      o.textContent = socks5LanIPs[i] + ' only';
      sel.appendChild(o);
    }
    sel.setAttribute('data-lan', '1');
  }
  try {
    var saved = JSON.parse(localStorage.getItem('akira.s5') || '{}');
    if (saved.bind && sel) {
      for (var j = 0; j < sel.options.length; j++) {
        if (sel.options[j].value === saved.bind) { sel.value = saved.bind; break; }
      }
    }
    if (saved.portMode) document.getElementById('socks5-port-mode').value = saved.portMode;
    if (saved.port) document.getElementById('socks5-port').value = saved.port;
    if (typeof saved.fb === 'boolean') document.getElementById('socks5-fallback').checked = saved.fb;
    if (saved.user) document.getElementById('socks5-user').value = saved.user;
    if (saved.pass) document.getElementById('socks5-pass').value = saved.pass;
  } catch (e) {}
  socks5PortModeChange();
  socks5Poll();
}

function socks5PortModeChange() {
  var mode = document.getElementById('socks5-port-mode').value;
  document.getElementById('socks5-port-wrap').style.display = mode === 'manual' ? '' : 'none';
  document.getElementById('socks5-fb-wrap').style.display = mode === 'manual' ? '' : 'none';
}

function socks5SaveSettings() {
  try {
    localStorage.setItem('akira.s5', JSON.stringify({
      bind: document.getElementById('socks5-bind').value,
      portMode: document.getElementById('socks5-port-mode').value,
      port: document.getElementById('socks5-port').value.trim(),
      fb: document.getElementById('socks5-fallback').checked,
      user: document.getElementById('socks5-user').value,
      pass: document.getElementById('socks5-pass').value
    }));
  } catch (e) {}
}

async function socks5Poll() {
  try {
    var res = await fetch('/api/socks5/status');
    var d = await res.json().catch(function(){ return {}; });
    socks5Render(d);
  } catch (e) {}
}
function socks5Render(d) {
  var running = !!d.running;
  var clients = Array.isArray(d.clients) ? d.clients : [];
  var btn = document.getElementById('socks5-toggle');
  if (btn) {
    if (running) {
      btn.className = 'mem-btn-stop';
      btn.innerHTML = '<i class="fa-solid fa-stop"></i> Stop';
    } else {
      btn.className = 'mem-btn-run';
      btn.innerHTML = '<i class="fa-solid fa-play"></i> Start';
    }
  }
  var ids = ['socks5-bind', 'socks5-port-mode', 'socks5-port', 'socks5-fallback', 'socks5-user', 'socks5-pass'];
  for (var i = 0; i < ids.length; i++) {
    var el = document.getElementById(ids[i]);
    if (el) el.disabled = running;
  }

  var conn = document.getElementById('socks5-conn');
  if (conn) {
    if (running) {
      var port = d.port || 0;
      var usable = (d.bind && d.bind !== '0.0.0.0' && d.bind !== '::') ? [d.bind] : socks5LanIPs;
      if (!usable.length) usable = ['127.0.0.1'];
      var passInp = document.getElementById('socks5-pass');
      var passVal = passInp ? passInp.value : '';
      var authDisp = d.auth_enabled ? (d.auth_user || 'user') + ':•••@' : '';
      var authCopy = d.auth_enabled ? (d.auth_user || 'user') + (passVal ? ':' + passVal : '') + '@' : '';
      document.getElementById('socks5-conn-str').textContent = authDisp + usable[0] + ':' + port;
      socks5ConnStr = authCopy + usable[0] + ':' + port;
      var alt = '';
      if (d.auth_enabled) alt = 'Auth required (user "' + (d.auth_user || '') + '")';
      if (usable.length > 1) alt += (alt ? ' · also: ' : 'also: ') + usable.slice(1).map(function(ip){ return ip + ':' + port; }).join('  ');
      document.getElementById('socks5-conn-alt').textContent = alt;
      conn.style.display = '';
    } else {
      conn.style.display = 'none';
      socks5ConnStr = '';
    }
  }

  var list = document.getElementById('socks5-list');
  if (list) {
    if (!running) {
      list.innerHTML = '<div class="sxt-empty">Server is stopped</div>';
    } else if (!clients.length) {
      list.innerHTML = '<div class="sxt-empty">No clients connected</div>';
    } else {
      var html = '';
      for (var i = 0; i < clients.length; i++) {
        html += '<div class="s5-peer"><i class="fa-solid fa-plug"></i><span>' + sxEsc(clients[i]) + '</span></div>';
      }
      list.innerHTML = html;
    }
  }
  var st = document.getElementById('socks5-status');
  if (st) st.textContent = running ? ('Listening on ' + (d.addr || '') + ' — ' + clients.length + ' client(s)') : 'Stopped';
}
function socks5CopyConn() {
  if (!socks5ConnStr) return;
  if (navigator.clipboard) {
    navigator.clipboard.writeText(socks5ConnStr).then(
      function(){ tmToast('fa-copy', 'Proxy address copied'); },
      function(){ tmToast('fa-triangle-exclamation', 'Copy failed'); }
    );
  }
}

async function socks5Toggle() {
  var btn = document.getElementById('socks5-toggle');
  var running = btn && btn.classList.contains('mem-btn-stop');
  try {
    if (running) {
      await fetch('/api/socks5/stop', { method: 'POST' });
      tmToast('fa-server', 'SOCKS5 server stopped');
    } else {
      var body = {
        bind: document.getElementById('socks5-bind').value,
        user: document.getElementById('socks5-user').value.trim(),
        pass: document.getElementById('socks5-pass').value
      };
      if (document.getElementById('socks5-port-mode').value === 'manual') {
        var p = parseInt(document.getElementById('socks5-port').value.trim(), 10);
        if (!p || p < 1 || p > 65535) {
          await cmAlert('Enter a valid port (1-65535) or switch Port to Auto.', 'SOCKS5 Proxy');
          return;
        }
        body.port = p;
        body.strict = !document.getElementById('socks5-fallback').checked;
      }
      socks5SaveSettings();
      var res = await fetch('/api/socks5/start', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(body)
      });
      var d = await res.json().catch(function(){ return {}; });
      if (!res.ok || d.error) { await cmAlert('Start failed: ' + (d.error || res.status), 'SOCKS5 Proxy'); return; }
      tmToast('fa-server', 'SOCKS5 listening on port ' + d.port + (d.fell_back ? ' (requested port busy — fell back)' : ''));
    }
  } catch (e) { await cmAlert('Request failed: ' + e.message, 'SOCKS5 Proxy'); }
  socks5Poll();
}

      kids.setAttribute('data-loaded', '1');
      regLoadChildren(row, kids);
    }
  } else {
    kids.style.display = 'none';
    if (caret) caret.classList.remove('open');
  }
}

async function regLoadChildren(row, kids) {
  var root = row.getAttribute('data-root');
  var path = row.getAttribute('data-path');
  kids.innerHTML = '<div class="sxt-empty" style="padding:6px 16px;text-align:left">Loading…</div>';
  try {
    var res = await fetch('/api/reg/keys?root=' + encodeURIComponent(root) + '&path=' + encodeURIComponent(path));
    var d = await res.json().catch(function(){ return {}; });
    if (d.error) { kids.innerHTML = '<div class="sxt-empty" style="padding:6px 16px;text-align:left">' + sxEsc(d.error) + '</div>'; return; }
    var keys = d.keys || [];
    var caret = row.querySelector('.reg-caret');
    if (!keys.length) {
      kids.innerHTML = '';
      kids.style.display = 'none';
      if (caret) { caret.classList.add('leaf'); caret.classList.remove('open'); }
      return;
    }
    if (caret) caret.classList.remove('leaf');
    var html = '';
    for (var i = 0; i < keys.length; i++) {
      var childPath = path ? (path + '\\' + keys[i]) : keys[i];
      html += regNodeHtml(root, keys[i], childPath);
    }
    kids.innerHTML = html;
  } catch (e) {
    kids.innerHTML = '<div class="sxt-empty" style="padding:6px 16px;text-align:left">Request failed</div>';
  }
}

async function botApi(path, body) {
  var opt = body ? { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) } : {};
  var r = await fetch('/api/bot' + path, opt);
  var text = await r.text();
  try {
    return text ? JSON.parse(text) : {};
  } catch (e) {
    throw new Error('HTTP ' + r.status + ' — invalid server response: ' + text.slice(0, 150));
  }
}

async function botLoadSessions() {
  try {
    var r = await botApi('/sessions');
    if (r.ok) { botSessionsCache = r.sessions || []; botRenderSessions(); }
  } catch (e) {}
}

function botBadge(state) {
  if (state === 'ready') return 'ready';
  if (state === 'dead') return 'dead';
  return 'busy';
}

function botRenderSessions() {
  var list = document.getElementById('bot-sess-list');
  list.innerHTML = '';
  if (!botSessionsCache.length) {
    list.innerHTML = '<span class="mem-hint">No hidden browsers running — start one below.</span>';
    return;
  }
  botSessionsCache.forEach(function(s) {
    var div = document.createElement('div');
    div.className = 'bot-sess';
    div.innerHTML =
      '<div class="bot-sess-head">' +
        '<span class="bot-sess-label">' + esc(s.label) + '</span>' +
        '<span class="bot-badge bot-badge-' + botBadge(s.state) + '">' + esc(s.state) + '</span>' +
      '</div>' +
      '<div class="bot-sess-meta">' + esc(s.browser) + ' · pid ' + s.pid + (s.cookie_count ? ' · ' + s.cookie_count + ' cookies' : '') + '</div>' +
      '<div class="bot-sess-btns">' +
        '<button class="tbtn" type="button" data-act="live" title="Watch and control on the hidden desktop"><i class="fa-solid fa-tower-broadcast"></i> Live</button>' +
        '<button class="tbtn" type="button" data-act="stop" title="Close this browser"><i class="fa-solid fa-stop"></i> Stop</button>' +

    div.querySelector('[data-act=stop]').onclick = function(ev) { ev.stopPropagation(); botStop(s.id); };
    list.appendChild(div);
  });
}

  try {
    var r = await botApi('/cookies/dump', {});
    var list = document.getElementById('bot-dump-list');

    if (r.ok && r.profiles && r.profiles.length) {
      r.profiles.forEach(function(p) {

          '<button class="mem-btn-run bot-dump-start" type="button"><i class="fa-solid fa-play"></i> Start</button>';
        row.querySelector('.bot-dump-vault').onclick = function(ev) {

          b.disabled = true;

              botVaultCache.unshift(r.entry);
              botVaultRender();
              tmToast('fa-vault', 'Saved ' + (r.entry.cookie_count || 0) + ' cookies to vault');
            } else cmAlert('Vault save failed: ' + (r.error || '?'), 'Cookie Vault');
          }).finally(function() { b.disabled = false; });
        };
        row.querySelector('.bot-dump-start').onclick = function(ev) {
          var b = ev.currentTarget;
          b.disabled = true;
          botStartSession(p.browser, p.browser + ' ' + p.profile, p.json).finally(function() { b.disabled = false; });
        };
        list.appendChild(row);
      });
    } else {
      list.innerHTML = '<span class="mem-hint">' + esc(r.error || 'No cookies recovered.') + '</span>';
    }
  } catch (e) {
    document.getElementById('bot-dump-list').innerHTML = '<span class="mem-hint">Dump failed: ' + esc(String(e)) + '</span>';
  }
  btn.disabled = false;
  btn.innerHTML = '<i class="fa-solid fa-magnifying-glass"></i> Scan';
}

var botVaultCache = [];

async function botVaultLoad() {
  try {
    var r = await botApi('/vault');

function botVaultRender() {
  var el = document.getElementById('bot-vault-list');
  if (!el) return;
  el.innerHTML = '';
  if (!botVaultCache.length) {
    el.innerHTML = '<span class="mem-hint">Vault empty. Scan browsers above or import a cookies file.</span>';
    return;
  }
  botVaultCache.forEach(function(v) {
    var row = document.createElement('div');
    row.className = 'bot-dump-row';
    var host = (v.domains && v.domains.length) ? (' · ' + v.domains.slice(0, 2).join(', ')) : '';
    var when = (v.saved_at || '').slice(0, 16).replace('T', ' ');
    row.innerHTML = '<span style="flex:1;overflow:hidden;text-overflow:ellipsis;white-space:nowrap" title="' + esc(v.label) + host + '">' + esc(v.label) + host + '</span>' +
      '<span class="bot-sess-meta">' + (v.cookie_count || 0) + ' · ' + esc(when) + '</span>' +
      '<button class="mem-btn-run" data-t="start" type="button" title="New hidden browser with these cookies"><i class="fa-solid fa-play"></i></button>' +
      '<button class="tbtn" data-t="dl" type="button" title="Download cookies file"><i class="fa-solid fa-download"></i></button>' +
      '<button class="tbtn" data-t="del" type="button" title="Delete"><i class="fa-solid fa-trash"></i></button>';
    row.querySelector('[data-t=start]').onclick = function() { botVaultStart(v.id, v.label); };
    row.querySelector('[data-t=dl]').onclick = function() { botVaultExport(v.id); };
    row.querySelector('[data-t=del]').onclick = function() { botVaultDelete(v.id); };

  });

async function botVaultStart(id, name) {
  var r = await botApi('/vault/start', { id: id });
  if (!r.ok) { cmAlert('Start failed: ' + (r.error || '?'), 'Hidden Browsers'); return; }
  if (r.warning) tmToast('fa-triangle-exclamation', r.warning);
  await botLoadSessions();
  tmToast('fa-vault', 'Hidden browser started: ' + name);
}

async function botVaultDelete(id) {
  if (!await cmConfirm('Delete this cookie set?', 'Cookie Vault')) return;
  await botApi('/vault/delete', { id: id });
  botVaultCache = botVaultCache.filter(function(v) { return v.id !== id; });
  botVaultRender();
}

async function botVaultExport(id) {
  var r = await botApi('/vault/export?id=' + encodeURIComponent(id));
  if (!r.ok) { cmAlert('Export failed: ' + (r.error || '?'), 'Cookie Vault'); return; }
  var blob = new Blob([r.data || ''], { type: 'application/octet-stream' });
  var a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = r.filename || ((r.label || 'cookies') + '.txt');
  document.body.appendChild(a);
  a.click();
  setTimeout(function() { URL.revokeObjectURL(a.href); a.remove(); }, 500);
}

async function botVaultImportFile(inp) {
  var f = inp.files && inp.files[0];
  inp.value = '';
  if (!f) return;
  if (f.size > 3 * 1024 * 1024) { cmAlert('File too large (max 3 MB).', 'Cookie Vault'); return; }
  var text = await f.text();
  var r = await botApi('/vault/import', { label: f.name.replace(/\.(json|txt)$/i, ''), data: text });
  if (!r.ok) { cmAlert('Import failed: ' + (r.error || '?'), 'Cookie Vault'); return; }
  botVaultCache.unshift(r.entry);
  botVaultRender();
  tmToast('fa-vault', 'Imported ' + (r.entry.cookie_count || 0) + ' cookies');
}

function fpOpen(id) {
  var el = document.getElementById('fp-' + id);
  if (!el.classList.contains('open')) {
    el.classList.add('open');
    if (!fpRestorePos(el)) setTimeout(function() { fpCenter(el); }, 10);
  }
  fpFocus(el);
  var btn = document.getElementById('btn-' + id);
  if (btn) btn.classList.add('modal-active');
  if (id === 'proc')  { loadProcs(); procAutoStart(); }
  if (id === 'files') WE.init();
  if (id === 'term'  && !termWs) termConnect();
  if (id === 'keys')  keysStart();
  if (typeof tmRefreshActive === 'function') tmRefreshActive();
}

  if (el) el.classList.remove('open');
  var btn = document.getElementById('btn-' + id);
  if (btn) btn.classList.remove('modal-active');

}

function fpToggle(id) {
  var el = document.getElementById('fp-' + id);
  if (!el.classList.contains('open')) {
    fpOpen(id);
  } else if (!fpIsTop(el)) {
    fpFocus(el);
  } else {
    fpClose(id);
  }
}

var fpZTop = 7000;
function fpFocus(el) {
  fpZTop++;
  el.style.zIndex = fpZTop;
}
function fpIsTop(el) {
  var myZ = parseInt(el.style.zIndex || '7000', 10);
  var els = document.querySelectorAll('.fp-modal.open');
  for (var i = 0; i < els.length; i++) {
    if (els[i] !== el) {
      var z = parseInt(els[i].style.zIndex || '7000', 10);
      if (z > myZ) return false;
    }
  }
  return true;
}

/* ── Modal position persistence (per modal id, survives reloads) ── */
function fpSavePos(fp) {
  try {
    var pos = JSON.parse(localStorage.getItem('akira.modalpos') || '{}');
    pos[fp.id] = { left: fp.style.left, top: fp.style.top, width: fp.style.width, height: fp.style.height };
    localStorage.setItem('akira.modalpos', JSON.stringify(pos));
  } catch (_) {}
}
function fpRestorePos(fp) {
  try {
    var pos = JSON.parse(localStorage.getItem('akira.modalpos') || '{}');
    var p = pos[fp.id];
    if (!p) return false;
    var l = parseInt(p.left, 10), t = parseInt(p.top, 10);
    if (isNaN(l) || isNaN(t) || l > window.innerWidth - 80 || t > window.innerHeight - 40 || t < 0) return false;
    fp.style.left = p.left; fp.style.top = p.top;
    if (p.width) fp.style.width = p.width;
    if (p.height) fp.style.height = p.height;
    return true;
  } catch (_) { return false; }

  /* ── uploads: small files go up in one multipart shot; anything larger
        than UPLOAD_CHUNK is streamed in 8MB chunks with offset writes ── */
  var UPLOAD_CHUNK=8*1024*1024;
  function uploadFiles(files) {
    if(!files||!files.length)return;
    var arr=Array.from(files),idx=0,totalBytes=0,i;
    for(i=0;i<arr.length;i++)totalBytes+=arr[i].size;
    var doneBytes=0,okCount=0;
    var bar=document.getElementById('w11-upload-bar');
    var wrap=document.getElementById('w11-upload-bar-wrap');
    if(wrap)wrap.classList.add('active');
    if(bar)bar.style.width='0%';
    function prog(){if(bar&&totalBytes>0)bar.style.width=Math.min(100,Math.round(doneBytes/totalBytes*100))+'%';}
    function finish(){
      setTimeout(function(){
        if(wrap)wrap.classList.remove('active');
        if(bar)bar.style.width='0%';
      },400);
      refresh();
      weToast('fa-arrow-up-from-bracket','Uploaded '+okCount+'/'+arr.length+' file(s)');
    }
    function uploadSimple(f,onDone,onErr){
      var fd=new FormData();fd.append('file',f);
      fetch('/api/files/upload?path='+encodeURIComponent(_path),{method:'POST',body:fd})
        .then(function(r){if(!r.ok)throw new Error('HTTP '+r.status);doneBytes+=f.size;prog();onDone();})
        .catch(onErr);
    }
    function uploadChunked(f,onDone,onErr){
      var id='u'+Date.now().toString(36)+Math.random().toString(36).slice(2,8);
      var off=0;
      (function step(){
        var end=Math.min(off+UPLOAD_CHUNK,f.size);
        fetch('/api/files/upload_chunk?path='+encodeURIComponent(_path)+'&name='+encodeURIComponent(f.name)+
              '&offset='+off+'&total='+f.size+'&id='+id,{method:'POST',body:f.slice(off,end)})
          .then(function(r){
            if(!r.ok)return r.text().then(function(t){throw new Error(t);});
            doneBytes+=end-off;off=end;prog();
            if(off<f.size)step();else onDone();
          }).catch(onErr);
      })();
    }
    (function next(){
      if(idx>=arr.length){finish();return;}
      var f=arr[idx];
      var onErr=function(e){doneBytes+=f.size;weToast('fa-triangle-exclamation','Upload error: '+f.name);idx++;next();};
      var onDone=function(){okCount++;idx++;next();};
      if(f.size>UPLOAD_CHUNK)uploadChunked(f,onDone,onErr);
      else uploadSimple(f,onDone,onErr);
    })();
  }

  /* ── Download selection as a zip, streamed from the server (no archive is
        materialised on the target's disk) ── */
  function downloadZip(){
    var sel=Array.from(_sel);
    if(!sel.length)return;
    var name=(sel.length===1?sel[0].replace(/[\\\/]+$/,'').split(/[\\\/]/).pop():'Archive')+'.zip';
    var bar=document.getElementById('w11-upload-bar');
    var wrap=document.getElementById('w11-upload-bar-wrap');
    weToast('fa-file-arrow-down','Preparing archive…');
    fetch('/api/files/zipdl?paths='+encodeURIComponent(sel.join(',')))
      .then(function(r){
        if(!r.ok)return r.text().then(function(t){throw new Error(t);});
        var rd=r.body.getReader(),chunks=[],got=0;
        if(wrap)wrap.classList.add('active');
        function pump(){
          return rd.read().then(function(res){
            if(res.done)return;
            chunks.push(res.value);got+=res.value.length;
            if(bar)bar.style.width=((Math.round(got/1048576))%100)+'%';
            return pump();
          });
        }
        return pump().then(function(){
          var blob=new Blob(chunks,{type:'application/zip'});
          var a=document.createElement('a');
          a.href=URL.createObjectURL(blob);a.download=name;
          document.body.appendChild(a);a.click();a.remove();
          setTimeout(function(){URL.revokeObjectURL(a.href);},20000);
          weToast('fa-check','Downloaded '+name);
        });
      })
      .catch(function(e){weToast('fa-triangle-exclamation','Error: '+e.message);})
      .finally(function(){if(wrap)wrap.classList.remove('active');if(bar)bar.style.width='0%';});
  }

    var newName=stem+' - Copy'+suffix;

// ═══ Terminal — ConPTY session with a fixed-grid VT emulator ═══════════════
// The server spawns the shell attached to a Windows pseudo console, so this
// client implements real terminal semantics: a screen-sized cell grid with
// scrollback, scroll regions, alt screen, SGR (8/16/256/RGB), insert/delete,
// OSC titles and DA/DSR replies. Keystrokes go straight to the PTY as VT.
let termWs = null;
let termCdPath = null;
let termLastDir = null;
let termEmu = null;
let termCellSize = null;
let termRenderPending = false;
let termFocused = false;
const termOut = document.getElementById('term-out');

function termCdConsume() {
  // Live "open terminal here": cd the running shell. (The initial cwd is
  // passed to the server as a spawn parameter in termConnect.)
  if (termCdPath && termWs && termWs.readyState === WebSocket.OPEN) {
    termLastDir = termCdPath;
    termWs.send(JSON.stringify({type: 'input', data: 'cd "' + termCdPath + '"\r\n'}));
    termCdPath = null;
  }
}

const TERM_FG     = ['#0c0c0c','#c50f1f','#13a10e','#c19c00','#0037da','#881798','#3a96dd','#cccccc'];
const TERM_FG_BR  = ['#767676','#e74856','#16c60c','#f9f1a5','#3b78ff','#b4009e','#61d6d6','#f2f2f2'];
const TERM_DEF_FG = '#cccccc';
const TERM_DEF_BG = '#0c0c0c';
const TERM_DEC_GFX = {'\u0060':'◆','a':'▒','f':'°','g':'±','j':'┘','k':'┐','l':'┌','m':'└','n':'┼','q':'─','t':'├','u':'┤','v':'┴','w':'┬','x':'│','y':'≤','z':'≥','{':'π','|':'≠','}':'£','~':'·'};

function termColor256(n) {
  n = ((n % 256) + 256) % 256;
  if (n < 16) return (n < 8 ? TERM_FG : TERM_FG_BR)[n % 8];
  if (n < 232) {
    const lv = [0x00, 0x5f, 0x87, 0xaf, 0xd7, 0xff], i = n - 16;
    return 'rgb(' + lv[(i / 36) | 0] + ',' + lv[((i / 6) | 0) % 6] + ',' + lv[i % 6] + ')';
  }
  const g = 8 + 10 * (n - 232);
  return 'rgb(' + g + ',' + g + ',' + g + ')';
}

function TermEmu(cols, rows) {
  var e = {
    cols: cols, rows: rows, grid: [], sb: [],
    cx: 0, cy: 0, fg: TERM_DEF_FG, bg: null,
    bold: false, dim: false, italic: false, under: false, strike: false, reverse: false,
    wrapPending: false, wrapMode: true, cursorVis: true,
    st: 0, sbm: rows - 1, alt: false, savedMain: null,
    gfxNext: false, gfx: false, bracketed: false, saved: null, saved2: null
  };
  termGridReset(e);
  return e;
}
function termBlank(e) { var l = new Array(e.cols); for (var i = 0; i < e.cols; i++) l[i] = { c: ' ', s: '' }; return l; }
function termGridReset(e) { e.grid = []; for (var y = 0; y < e.rows; y++) e.grid.push(termBlank(e)); e.cx = 0; e.cy = 0; e.wrapPending = false; }
function termCssOf(e) {
  var f = e.fg, b = e.bg;
  if (e.reverse) { var t = f; f = b || TERM_DEF_BG; b = t || TERM_DEF_FG; }
  var p = [];
  if (f) p.push('color:' + f);
  if (b) p.push('background-color:' + b);
  if (e.bold) p.push('font-weight:600');
  if (e.dim) p.push('opacity:.55');
  if (e.italic) p.push('font-style:italic');
  var d = (e.under ? ' underline' : '') + (e.strike ? ' line-through' : '');
  if (d) p.push('text-decoration:' + d.trim());
  return p.join(';');
}
function termScrollUp(e, n) {
  n = Math.min(n, e.sbm - e.st + 1);
  for (var i = 0; i < n; i++) {
    var top = e.grid.splice(e.st, 1)[0];
    e.grid.splice(e.sbm, 0, termBlank(e));
    if (!e.alt && e.st === 0 && e.sbm === e.rows - 1) {
      e.sb.push(top);
      if (e.sb.length > 2000) e.sb.splice(0, e.sb.length - 2000);
    }
  }
}
function termScrollDown(e, n) {
  n = Math.min(n, e.sbm - e.st + 1);
  for (var i = 0; i < n; i++) { e.grid.splice(e.sbm, 1); e.grid.splice(e.st, 0, termBlank(e)); }
}
function termNewline(e) {
  e.wrapPending = false;
  if (e.cy === e.sbm) termScrollUp(e, 1);
  else e.cy = Math.min(e.cy + 1, e.rows - 1);
}
function termPut(e, ch) {
  if (e.gfx && TERM_DEC_GFX[ch]) ch = TERM_DEC_GFX[ch];
  if (e.wrapPending) { e.cx = 0; termNewline(e); e.wrapPending = false; }
  e.grid[e.cy][e.cx] = { c: ch, s: termCssOf(e) };
  if (e.cx === e.cols - 1) { if (e.wrapMode) e.wrapPending = true; }
  else e.cx++;
}
function termAltScreen(e, on, withCursor) {
  if (on === e.alt) return;
  if (on) {
    e.savedMain = { grid: e.grid, cx: e.cx, cy: e.cy, st: e.st, sbm: e.sbm };
    termGridReset(e);
    e.st = 0; e.sbm = e.rows - 1; e.alt = true;
  } else if (e.savedMain) {
    var m = e.savedMain;
    e.grid = m.grid; e.cx = m.cx; e.cy = m.cy; e.st = m.st; e.sbm = m.sbm; e.alt = false;
  }
}

function termEd(e, mode) {
  if (mode === 2 || mode === 3) { termGridReset(e); if (mode === 3) e.sb = []; return; }
  var line, x, y;
  if (mode === 0) {
    line = e.grid[e.cy];
    for (x = e.cx; x < e.cols; x++) line[x] = { c: ' ', s: '' };
    for (y = e.cy + 1; y < e.rows; y++) e.grid[y] = termBlank(e);
  } else if (mode === 1) {
    line = e.grid[e.cy];
    for (x = 0; x <= e.cx; x++) line[x] = { c: ' ', s: '' };
    for (y = 0; y < e.cy; y++) e.grid[y] = termBlank(e);
  }
}
function termEl(e, mode) {
  var line = e.grid[e.cy];
  var from = mode === 1 ? 0 : e.cx, to = mode === 1 ? e.cx : e.cols - 1;
  if (mode === 2) { from = 0; to = e.cols - 1; }
  for (var x = from; x <= to; x++) line[x] = { c: ' ', s: '' };
}
function termInsLines(e, n) {
  if (e.cy < e.st || e.cy > e.sbm) return;
  n = Math.min(n, e.sbm - e.cy + 1);
  for (var i = 0; i < n; i++) { e.grid.splice(e.cy, 0, termBlank(e)); e.grid.splice(e.sbm + 1, 1); }
}
function termDelLines(e, n) {
  if (e.cy < e.st || e.cy > e.sbm) return;
  n = Math.min(n, e.sbm - e.cy + 1);
  for (var i = 0; i < n; i++) { e.grid.splice(e.cy, 1); e.grid.splice(e.sbm, 0, termBlank(e)); }
}
function termInsChars(e, n) {
  var line = e.grid[e.cy];
  for (var i = 0; i < n; i++) { line.splice(e.cx, 0, { c: ' ', s: '' }); line.pop(); }
}
function termDelChars(e, n) {
  var line = e.grid[e.cy];
  n = Math.min(n, e.cols - e.cx);
  for (var i = 0; i < n; i++) { line.splice(e.cx, 1); line.push({ c: ' ', s: '' }); }
}
function termResizeEmu(e, cols, rows) {
  if (cols === e.cols && rows === e.rows) return;
  var ng = [];
  for (var y = 0; y < rows; y++) {
    var l = (e.grid[y] || []).slice(0, cols);
    while (l.length < cols) l.push({ c: ' ', s: '' });
    ng.push(l);
  }
  e.grid = ng; e.cols = cols; e.rows = rows;
  e.st = 0; e.sbm = rows - 1;
  e.cx = Math.min(e.cx, cols - 1); e.cy = Math.min(e.cy, rows - 1);
  e.wrapPending = false;
}

// ── VT parser ─────────────────────────────────────────────────────────────
function termSgr(e, ps) {
  for (var k = 0; k < ps.length; k++) {
    var v = parseInt(ps[k], 10); if (isNaN(v)) v = 0;
    if (v === 0) { e.fg = TERM_DEF_FG; e.bg = null; e.bold = e.dim = e.italic = e.under = e.strike = e.reverse = false; }
    else if (v === 1) e.bold = true;
    else if (v === 2) e.dim = true;
    else if (v === 3) e.italic = true;
    else if (v === 4) e.under = true;
    else if (v === 7) e.reverse = true;
    else if (v === 9) e.strike = true;
    else if (v === 22) { e.bold = false; e.dim = false; }
    else if (v === 23) e.italic = false;
    else if (v === 24) e.under = false;
    else if (v === 27) e.reverse = false;
    else if (v === 29) e.strike = false;
    else if (v >= 30 && v <= 37) e.fg = TERM_FG[v - 30];
    else if (v === 38) {
      if (ps[k+1] === '5') { e.fg = termColor256(parseInt(ps[k+2], 10) || 0); k += 2; }
      else if (ps[k+1] === '2') { e.fg = 'rgb(' + (parseInt(ps[k+2],10)||0) + ',' + (parseInt(ps[k+3],10)||0) + ',' + (parseInt(ps[k+4],10)||0) + ')'; k += 4; }
    }
    else if (v === 39) e.fg = TERM_DEF_FG;
    else if (v >= 40 && v <= 47) e.bg = TERM_FG[v - 40];
    else if (v === 48) {
      if (ps[k+1] === '5') { e.bg = termColor256(parseInt(ps[k+2], 10) || 0); k += 2; }
      else if (ps[k+1] === '2') { e.bg = 'rgb(' + (parseInt(ps[k+2],10)||0) + ',' + (parseInt(ps[k+3],10)||0) + ',' + (parseInt(ps[k+4],10)||0) + ')'; k += 4; }
    }
    else if (v === 49) e.bg = null;
    else if (v >= 90 && v <= 97) e.fg = TERM_FG_BR[v - 90];
    else if (v >= 100 && v <= 107) e.bg = TERM_FG_BR[v - 100];
  }
}
function termCsi(e, fin, ps, p0, p1, priv) {
  switch (fin) {
    case 'm': termSgr(e, ps); return '';
    case 'A': e.cy -= p0 || 1; break;
    case 'B': e.cy += p0 || 1; break;
    case 'C': e.cx += p0 || 1; break;
    case 'D': e.cx -= p0 || 1; break;
    case 'E': e.cy += p0 || 1; e.cx = 0; break;
    case 'F': e.cy -= p0 || 1; e.cx = 0; break;
    case 'G': case '\u0060': e.cx = (p0 || 1) - 1; break;
    case 'd': e.cy = (p0 || 1) - 1; break;
    case 'H': case 'f': e.cy = (p0 || 1) - 1; e.cx = (p1 || 1) - 1; break;
    case 'J': termEd(e, p0); break;
    case 'K': termEl(e, p0); break;
    case 'L': termInsLines(e, p0 || 1); break;
    case 'M': termDelLines(e, p0 || 1); break;
    case 'P': termDelChars(e, p0 || 1); break;
    case '@': termInsChars(e, p0 || 1); break;
    case 'X': termEraseChars(e, p0 || 1); break;
    case 'S': termScrollUp(e, p0 || 1); break;
    case 'T': termScrollDown(e, p0 || 1); break;
    case 'r': {
      var top = p0 || 1, bot = p1 || e.rows;
      if (top >= bot) { e.st = 0; e.sbm = e.rows - 1; }
      else { e.st = Math.max(0, top - 1); e.sbm = Math.min(e.rows - 1, bot - 1); }
      e.cx = 0; e.cy = 0;
      break;
    }
    case 's': e.saved2 = { x: e.cx, y: e.cy }; break;
    case 'u': if (e.saved2) { e.cx = e.saved2.x; e.cy = e.saved2.y; } break;
    case 'h': case 'l': {
      var on = fin === 'h';
      for (var mi = 0; mi < ps.length; mi++) {
        var v = parseInt(ps[mi], 10);
        if (!priv) continue;
        if (v === 7) e.wrapMode = on;
        else if (v === 25) e.cursorVis = on;
        else if (v === 1049) termAltScreen(e, on, true);
        else if (v === 1047) termAltScreen(e, on, false);
        else if (v === 1048) { if (on) e.saved2 = { x: e.cx, y: e.cy }; else if (e.saved2) { e.cx = e.saved2.x; e.cy = e.saved2.y; } }
        else if (v === 2004) e.bracketed = on;
      }
      break;
    }
    case 'c': return '\x1b[?1;2c';                    // DA → VT100+AVO
    case 'n':
      if (p0 === 5) return '\x1b[0n';
      if (p0 === 6) return '\x1b[' + (e.cy + 1) + ';' + (e.cx + 1) + 'R';
      return '';
  }
  // Any cursor-addressing sequence cancels a pending line wrap.
  e.wrapPending = false;
  return '';
}
function termEraseChars(e, n) {
  var line = e.grid[e.cy];
  n = Math.min(n, e.cols - e.cx);
  for (var i = 0; i < n; i++) line[e.cx + i] = { c: ' ', s: '' };
}

function termFeed(e, text) {
  var i = 0, n = text.length, replies = '';
  while (i < n) {
    var ch = text[i];
    if (ch === '\x1b') {
      if (i + 1 >= n) break;
      var c1 = text[i + 1];
      if (c1 === '[') {
        var j = i + 2;
        while (j < n && text.charCodeAt(j) < 0x40) j++;
        if (j >= n) break; // incomplete CSI — wait for more data
        var fin = text[j], raw = text.slice(i + 2, j);
        var priv = raw.charAt(0) === '?';
        var ps = raw.replace(/^[?<>=%!]/, '').split(';');
        var p0 = parseInt(ps[0], 10) || 0;
        var p1 = ps.length > 1 ? (parseInt(ps[1], 10) || 0) : 0;
        replies += termCsi(e, fin, ps, p0, p1, priv);
        i = j + 1; continue;
      }
      if (c1 === ']') {
        var k = i + 2;
        while (k < n && text[k] !== '\x07' && !(text[k] === '\x1b' && k + 1 < n && text[k + 1] === '\\')) k++;
        if (k >= n) break; // incomplete OSC
        var om = text.slice(i + 2, k).match(/^(?:0|2);(.*)$/);
        if (om) { var tt = document.getElementById('term-tab-title'); if (tt) tt.textContent = om[1] || 'Terminal'; }
        i = text[k] === '\x07' ? k + 1 : k + 2; continue;
      }
      if (c1 === '(') { e.gfxNext = text[i + 2] === '0'; i += 3; continue; }
      if (c1 === ')') { i += 3; continue; }
      if (c1 === '7') { e.saved = { x: e.cx, y: e.cy, fg: e.fg, bg: e.bg, bold: e.bold, dim: e.dim, italic: e.italic, under: e.under, strike: e.strike, reverse: e.reverse }; i += 2; continue; }
      if (c1 === '8') {
        if (e.saved) { var s = e.saved; e.cx = s.x; e.cy = s.y; e.fg = s.fg; e.bg = s.bg; e.bold = s.bold; e.dim = s.dim; e.italic = s.italic; e.under = s.under; e.strike = s.strike; e.reverse = s.reverse; }
        i += 2; continue;
      }
      if (c1 === 'c') { var f = TermEmu(e.cols, e.rows); for (var key in f) e[key] = f[key]; i += 2; continue; }
      if (c1 === 'D') { termNewline(e); i += 2; continue; }
      if (c1 === 'E') { e.cx = 0; termNewline(e); i += 2; continue; }
      if (c1 === 'M') { if (e.cy === e.st) termScrollDown(e, 1); else if (e.cy > 0) e.cy--; i += 2; continue; }
      i += 2; continue; // other two-byte escapes
    }
    if (ch === '\n' || ch === '\x0b' || ch === '\x0c') { termNewline(e); i++; continue; }
    if (ch === '\r') { e.cx = 0; e.wrapPending = false; i++; continue; }
    if (ch === '\b') { if (e.cx > 0) e.cx--; i++; continue; }
    if (ch === '\t') { var tw = 8 - (e.cx % 8); for (var t = 0; t < tw; t++) termPut(e, ' '); i++; continue; }
    if (ch === '\x0e') { e.gfx = e.gfxNext; i++; continue; }
    if (ch === '\x0f') { e.gfx = false; i++; continue; }
    if (ch < ' ') { i++; continue; }
    termPut(e, ch); i++;
  }
  if (e.cx < 0) e.cx = 0; else if (e.cx >= e.cols) e.cx = e.cols - 1;
  if (e.cy < 0) e.cy = 0; else if (e.cy >= e.rows) e.cy = e.rows - 1;
  return replies;
}

// ── Renderer ──────────────────────────────────────────────────────────────
function termEscCell(ch) {
  return ch.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
}
function termRenderLine(line) {
  var html = '', curS = null, run = '';
  for (var c = 0; c < line.length; c++) {
    var cell = line[c];
    var tc = termEscCell(cell.c);
    if (cell.s !== curS) { if (run) html += curS ? '<span style="' + curS + '">' + run + '</span>' : run; curS = cell.s; run = ''; }
    run += tc;
  }
  if (run) html += curS ? '<span style="' + curS + '">' + run + '</span>' : run;
  return html;
}
function termRenderCursorLine(e) {
  var line = e.grid[e.cy];
  var at = line[e.cx];
  return termRenderLine(line.slice(0, e.cx)) +
         '<span class="term-cursor">' + (at.c === ' ' ? ' ' : termEscCell(at.c)) + '</span>' +
         termRenderLine(line.slice(e.cx + 1));
}
function termRender() {
  if (!termEmu) return;
  var e = termEmu, out = [];
  var sbs = e.sb, start = Math.max(0, sbs.length - 1000);
  var ft = termFindTerm ? termFindTerm.toLowerCase() : '';
  var addLine = function(line, html) {
    if (ft && termLineText(line).toLowerCase().indexOf(ft) >= 0) {
      out.push('<span class="term-find-hit">' + html + '</span>');
    } else {
      out.push(html);
    }
  };
  for (var i = start; i < sbs.length; i++) addLine(sbs[i], termRenderLine(sbs[i]));
  for (var y = 0; y < e.rows; y++) {
    addLine(e.grid[y], (y === e.cy && e.cursorVis && termFocused) ? termRenderCursorLine(e) : termRenderLine(e.grid[y]));
  }
  var keep = (termOut._stick === false) ? termOut.scrollTop : null;
  termOut.innerHTML = out.join('\n');
  if (keep !== null) termOut.scrollTop = Math.min(keep, termOut.scrollHeight);
  else termOut.scrollTop = termOut.scrollHeight;
}
function termWrite(t) {
  if (!termEmu) return;
  var replies = termFeed(termEmu, t);
  if (replies) termSend(replies);
  if (!termRenderPending) {
    termRenderPending = true;
    requestAnimationFrame(function() { termRenderPending = false; termRender(); });
  }
}
function termSend(s) {
  if (termWs && termWs.readyState === WebSocket.OPEN) termWs.send(JSON.stringify({type: 'input', data: s}));
}

// ── Connection & sizing ─────────────────────────────────────────────────────
function termGeometry() {
  if (!termCellSize) {
    var m = document.createElement('span');
    m.textContent = 'MMMMMMMMMM';
    m.style.cssText = 'position:absolute;visibility:hidden;white-space:pre';
    termOut.appendChild(m);
    var r = m.getBoundingClientRect();
    m.remove();
    termCellSize = { w: (r.width / 10) || 7.2, h: r.height || 17 };
  }
  var cols = Math.floor((termOut.clientWidth - 30) / termCellSize.w);
  var rows = Math.floor((termOut.clientHeight - 26) / termCellSize.h);
  return { cols: Math.max(40, Math.min(400, cols)), rows: Math.max(5, Math.min(120, rows)) };
}
/* ── Shell tabs: PowerShell / cmd.exe (backend: termShell) ── */
var termShellChoice = 'ps';
function termShellTab(which) {
  if (which === termShellChoice && termWs) return;
  termShellChoice = which;
  document.getElementById('term-tab-ps').classList.toggle('active', which === 'ps');
  document.getElementById('term-tab-cmd').classList.toggle('active', which === 'cmd');
  var tt = document.getElementById('term-tab-title');
  if (tt) tt.textContent = which === 'cmd' ? 'cmd' : 'PowerShell';
  termReconnect();
}

function termConnect() {
  if (termWs) return;
  var g = termGeometry();
  termEmu = TermEmu(g.cols, g.rows);
  var proto = location.protocol === 'https:' ? 'wss:' : 'ws:';
  var url = proto + '//' + location.host + '/ws/term?cols=' + g.cols + '&rows=' + g.rows;
  if (termShellChoice === 'cmd') url += '&shell=cmd';
  var dir = termCdPath || termLastDir;
  if (dir) url += '&dir=' + encodeURIComponent(dir);
  termCdPath = null;
  termWs = new WebSocket(url);
  var st = document.getElementById('term-stat');
  termWs.onopen = function() {
    st.innerHTML = '<i class="fa-solid fa-circle" style="font-size:7px"></i> Live';
    st.className = 'wt-stat on';
    termOut.focus();
  };
  termWs.onclose = function() {
    st.innerHTML = '<i class="fa-solid fa-circle" style="font-size:7px"></i>';
    st.className = 'wt-stat off';
    termWs = null;
    termWrite('\r\n\x1b[33m[disconnected — reconnect to respawn]\x1b[0m\r\n');
  };
  termWs.onmessage = function(ev) { termWrite(ev.data); };
}
function termReconnect() {
  if (termWs) { try { termWs.onclose = null; termWs.close(); } catch(_) {} termWs = null; }
  termConnect();
}
function termClear() {
  if (!termEmu) return;
  termEmu.sb = [];
  termGridReset(termEmu);
  termRender();
}

function termCopyAll() {
  if (!termEmu) { tmToast('fa-copy', 'Nothing to copy'); return; }
  var lines = [];
  var dump = function(line) {
    var s = '';
    for (var i = 0; i < line.length; i++) s += line[i].c;
    lines.push(s.replace(/\s+$/, ''));
  };
  termEmu.sb.forEach(dump);
  termEmu.grid.forEach(dump);
  if (navigator.clipboard) {

const TERM_KEYMAP = {
  Enter: '\r', Backspace: '\x7f', Tab: '\t', Escape: '\x1b',
  ArrowUp: '\x1b[A', ArrowDown: '\x1b[B', ArrowRight: '\x1b[C', ArrowLeft: '\x1b[D',
  Home: '\x1b[H', End: '\x1b[F', Delete: '\x1b[3~', Insert: '\x1b[2~',
  PageUp: '\x1b[5~', PageDown: '\x1b[6~',
  F1: '\x1bOP', F2: '\x1bOQ', F3: '\x1bOR', F4: '\x1bOS',
  F5: '\x1b[15~', F6: '\x1b[17~', F7: '\x1b[18~', F8: '\x1b[19~',
  F9: '\x1b[20~', F10: '\x1b[21~', F11: '\x1b[23~', F12: '\x1b[24~'
};
termOut.addEventListener('keydown', function(ev) {
  if (ev.ctrlKey && (ev.key === 'f' || ev.key === 'F')) {
    ev.preventDefault();
    termFindToggle();
    return;
  }
  if (!termWs || termWs.readyState !== WebSocket.OPEN) return;
  var k = ev.key;
  if (ev.ctrlKey) {
    if (k === 'c' || k === 'C') {
      var sel = window.getSelection();
      if (sel && sel.toString()) return; // selection exists → let the copy through
      termSend('\x03'); ev.preventDefault(); return;
    }
    if (k === 'v' || k === 'V' || k === 'Insert') return; // paste event handles it
    if (k.length === 1) {
      var code = k.toUpperCase().charCodeAt(0);
      if (code >= 64 && code <= 95) { termSend(String.fromCharCode(code - 64)); ev.preventDefault(); }
    }
    return;
  }
  if (ev.altKey) {
    if (k.length === 1) { termSend('\x1b' + k); ev.preventDefault(); }
    return;
  }
  if (k.length === 1) { termSend(k); ev.preventDefault(); return; }
  var seq = TERM_KEYMAP[k];
  if (seq) { termSend(seq); ev.preventDefault(); }
});
termOut.addEventListener('paste', function(ev) {
  ev.preventDefault();
  var t = (ev.clipboardData || window.clipboardData).getData('text');
  if (!t) return;
  t = t.replace(/\r\n/g, '\n').replace(/\n/g, '\r');
  if (termEmu && termEmu.bracketed) t = '\x1b[200~' + t + '\x1b[201~';
  termSend(t);
});

// ── Task Manager tab switcher (Processes / GPU / Services) ──
function tmProcTab(tab) {
  var panels = { proc: 'tm11-proc-panel', gpu: 'tm11-gpu-panel', svc: 'tm11-svc-panel' };
  var tabs   = { proc: 'tm11-tab-proc',   gpu: 'tm11-tab-gpu',   svc: 'tm11-tab-svc' };
  for (var k in panels) {
    var p = document.getElementById(panels[k]);
    var t = document.getElementById(tabs[k]);
    if (p) p.style.display = (k === tab) ? 'flex' : 'none';
    if (t) t.classList.toggle('tm11-tab-active', k === tab);
  }
  // GPU auto-refresh runs only while its tab is visible.
  if (gpuAutoTimer) { clearInterval(gpuAutoTimer); gpuAutoTimer = null; }
  if (procAutoTimer) { clearInterval(procAutoTimer); procAutoTimer = null; }
  if (tab === 'gpu')      loadGpu();
  else if (tab === 'svc') svcLoad();
  else                    { loadProcs(); procAutoStart(); }
}

// Processes auto-refresh (2s) while the Task Manager modal is open on the
// Processes tab; stops when the modal closes or another tab is selected.
var procAutoTimer = null;
function procAutoStart() {
  if (procAutoTimer) return;
  procAutoTimer = setInterval(function() {
    var m = document.getElementById('fp-proc');
    if (!m || !m.classList.contains('open')) {
      clearInterval(procAutoTimer); procAutoTimer = null; return;
    }
    // Don't clobber an in-progress kill or search typing with a stale load.
    loadProcs();
  }, 2000);
}

function _procIcon(name) {
  const n = name.toLowerCase();
  if (n === 'explorer.exe') return ['fa-folder', '#eab308'];
  if (['powershell.exe','windowsterminal.exe','cmd.exe','pwsh.exe'].includes(n)) return ['fa-terminal', '#4ec9b0'];
  if (n.match(/chrome|msedge|firefox|brave|opera/)) return ['fa-globe', '#60a5fa'];
  if (['code.exe','devenv.exe','notepad.exe','notepad++.exe','rider64.exe'].includes(n)) return ['fa-code', '#a78bfa'];
  if (['svchost.exe','services.exe','lsass.exe','wininit.exe'].includes(n)) return ['fa-gear', '#6b7280'];
  if (['system','registry','smss.exe','csrss.exe','winlogon.exe'].includes(n)) return ['fa-microchip', '#64748b'];
  if (n.match(/antivirus|defender|security|malware/)) return ['fa-shield-halved', '#22c55e'];
  if (n.endsWith('.exe')) return ['fa-window-maximize', '#3b82f6'];
  return ['fa-circle-dot', '#4b5563'];
}
function renderProcs() {
  const q = document.getElementById('proc-search').value.toLowerCase();
  let list = allProcs.filter(p => p.name.toLowerCase().includes(q) || String(p.pid).includes(q));
  list.sort((a, b) => {
    let va, vb;
    if      (pSortKey === 'pid')  { va=a.pid; vb=b.pid; }
    else if (pSortKey === 'name') { va=a.name.toLowerCase(); vb=b.name.toLowerCase(); }
    else if (pSortKey === 'cpu')  { va=parsePct(a.cpu); vb=parsePct(b.cpu); }
    else                          { va=parseMem(a.memory); vb=parseMem(b.memory); }
    return pSortAsc ? (va > vb ? 1 : -1) : (va < vb ? 1 : -1);
  });
  const wrap = document.getElementById('proc-tbl-wrap');
  wrap.innerHTML = '';
  const maxMem = Math.max(1, ...list.map(p => parseMem(p.memory)));
  list.forEach((p, ri) => {
    const [icoClass, icoColor] = _procIcon(p.name);
    const barPct = Math.round(parseMem(p.memory) / maxMem * 100);
    const barColor = barPct >= 80 ? '#ef4444' : barPct >= 50 ? '#f59e0b' : '#0078d4';
    const cpuPct = parsePct(p.cpu);
    const cpuBarPct = Math.min(100, Math.round(cpuPct));
    const cpuBarColor = cpuBarPct >= 80 ? '#ef4444' : cpuBarPct >= 50 ? '#f59e0b' : '#0078d4';
    const row = document.createElement('div');
    row.className = 'tm11-row row-anim';
    row.style.animationDelay = Math.min(ri * 7, 140) + 'ms';
    row.addEventListener('contextmenu', function(e) { procCtxMenu(e, p.pid, p.name); });
    row.innerHTML =
      '<div class="tm11-cell-name">' +
        '<i class="fa-solid ' + icoClass + ' tm11-proc-ico" style="color:' + icoColor + '"></i>' +
        '<span class="tm11-proc-name">' + esc(p.name) + '</span>' +
      '</div>' +
      '<div class="tm11-cell-pid">' + p.pid + '</div>' +
      '<div class="tm11-cell-cpu">' +
        '<div class="tm11-cpu-label">' + (p.cpu || '0%') + '</div>' +
        '<div class="tm11-cpu-track"><div class="tm11-cpu-fill" style="width:' + cpuBarPct + '%;background:' + cpuBarColor + '"></div></div>' +
      '</div>' +
      '<div class="tm11-cell-mem">' +
        '<div class="tm11-mem-label">' + p.memory + '</div>' +
        '<div class="tm11-mem-track"><div class="tm11-mem-fill" style="width:' + barPct + '%;background:' + barColor + '"></div></div>' +

  });

}

  if (cnt) cnt.textContent = 'Sampling GPU engines…';
  try {
    var d = await fetch('/api/gpu').then(function(r){return r.json();});

      if (cnt) cnt.textContent = 'GPU query failed: ' + d.error;

  } finally {
    gpuLoading = false;
  }
}

async function scanCams() {
  camStatus('Scanning…');
  try {
    const res = await fetch('/api/cams');
    if (!res.ok) { camStatus('Scan failed: server returned ' + res.status); return; }
    const ct = res.headers.get('content-type') || '';
    if (!ct.includes('application/json')) { camStatus('Scan timed out — try again'); return; }
    const cams = await res.json();
    const sel = document.getElementById('cam-select');
    sel.innerHTML = '<option value="">Select camera…</option>';
    if (!cams || cams.length === 0) { camStatus('No cameras found on remote machine'); return; }
    cams.forEach(c => {
      const o = document.createElement('option');
      o.value = String(c.index); o.textContent = c.name;
      sel.appendChild(o);
    });
    camStatus(cams.length + ' camera(s) found — select one and press Start');
    if (cams.length >= 1) sel.value = String(cams[0].index);
  } catch (e) { camStatus('Scan error: ' + e); }
}

let _camIndex = -1, _camName = '', _camStopped = true;
let _camRetryDelay = 1000, _camRetryTimer = null, _camWatchdog = null, _camLastFrame = 0;

function _camWatchdogStart() {
  _camWatchdogStop(); _camLastFrame = Date.now();
  _camWatchdog = setInterval(() => {
    if (Date.now() - _camLastFrame > 10000) { _camWatchdogStop(); if (!_camStopped) _camReconnect(); }
  }, 2000);
}
function _camWatchdogStop() { if (_camWatchdog) { clearInterval(_camWatchdog); _camWatchdog = null; } }
function _camReconnect() {
  if (_camStopped || _camIndex < 0) return;
  camStatus('Reconnecting…');
  if (_camRetryTimer) { clearTimeout(_camRetryTimer); _camRetryTimer = null; }
  _camRetryTimer = setTimeout(() => { _camRetryTimer = null; startCam(_camIndex, _camName); }, _camRetryDelay);
  _camRetryDelay = Math.min(_camRetryDelay * 2, 8000);
}

function startCam(index, name) {
  if (_camRetryTimer) { clearTimeout(_camRetryTimer); _camRetryTimer = null; }
  _camWatchdogStop();
  if (camWs) { try { camWs.onclose = null; camWs.close(); } catch(_){} camWs = null; }
  _camIndex = index; _camName = name; _camStopped = false; _camRetryDelay = 1000;
  camStatus('Connecting to "' + name + '"…');
  const proto = location.protocol === 'https:' ? 'wss:' : 'ws:';
  camWs = new WebSocket(proto + '//' + location.host + '/ws/cam');
  camWs.binaryType = 'arraybuffer';
  camWs.onopen = () => { camWs.send(JSON.stringify({index})); camShowCanvas(); camStatus('Streaming: ' + name); _camRetryDelay = 1000; _camWatchdogStart(); };
  camWs.onclose = () => { _camWatchdogStop(); camWs = null; if (_camStopped) { camHideCanvas(); camStatus('Stopped'); } else { _camReconnect(); } };
  camWs.onerror = () => { try { if (camWs) camWs.close(); } catch(_){} };
  camWs.onmessage = e => {
    if (typeof e.data === 'string') { try { const m = JSON.parse(e.data); if (m.error) camStatus('Server error: ' + m.error); } catch(_){} return; }
    const ab = e.data;
    if (!(ab instanceof ArrayBuffer) || ab.byteLength < 4) return;
    const hdr = new Uint8Array(ab, 0, 2);
    if (hdr[0] !== 0xFF || hdr[1] !== 0xD8) return;
    _camLastFrame = Date.now();
    createImageBitmap(new Blob([ab], {type:'image/jpeg'})).then(bmp => {
      if (camCanvas.width !== bmp.width || camCanvas.height !== bmp.height) { camCanvas.width = bmp.width; camCanvas.height = bmp.height; }
      camCtx.drawImage(bmp, 0, 0); bmp.close();
    }).catch(err => { camStatus('Frame decode error: ' + err.message); });
  };
}

function stopCam() {
  _camStopped = true;
  if (_camRetryTimer) { clearTimeout(_camRetryTimer); _camRetryTimer = null; }
  _camWatchdogStop();
  if (camWs) { try { camWs.onclose = null; camWs.close(); } catch(_){} camWs = null; }
  camHideCanvas(); camStatus('Stopped');
}

var MEM_PASSPHRASE = 'akira-mem-exec-v1';
// Encrypt the file bytes with AES-256-GCM in the browser before upload.
// Wire format = [12-byte IV][ciphertext + 16-byte tag]. Key = SHA-256(passphrase).
async function memEncryptPayload(file){
  if(!window.crypto||!window.crypto.subtle)throw new Error('Web Crypto unavailable (use localhost/https)');
  var raw=await file.arrayBuffer();
  var keyHash=await crypto.subtle.digest('SHA-256',new TextEncoder().encode(MEM_PASSPHRASE));
  var key=await crypto.subtle.importKey('raw',keyHash,{name:'AES-GCM'},false,['encrypt']);
  var iv=crypto.getRandomValues(new Uint8Array(12));
  var ct=await crypto.subtle.encrypt({name:'AES-GCM',iv:iv},key,raw);
  var out=new Uint8Array(iv.length+ct.byteLength);
  out.set(iv,0);
  out.set(new Uint8Array(ct),iv.length);
  return out;
}

(function() {
  var audWs        = null;
  var audCtx       = null;
  var deskFmt      = null;
  var micFmt       = null;
  var deskGain     = 0;
  var micGain      = 0;
  var deskNext     = { v: 0 };
  var micNext      = { v: 0 };
  var audRetryTimer = null;
  // WebCodecs AAC decoders (null when the source streams legacy PCM16).
  // Shape: {dec, key, sr, ts} — key detects format changes, ts is the µs
  // chunk timestamp counter (decoder only needs monotonicity).
  var deskDec      = null;
  var micDec       = null;

  function toggleAudioDrop() {
    var drop = document.getElementById('audio-drop');
    var btn  = document.getElementById('audio-btn');
    if (drop.classList.contains('open')) { drop.classList.remove('open'); return; }
    var r = btn.getBoundingClientRect();
    drop.style.top  = (r.bottom + 6) + 'px';
    drop.style.left = Math.max(8, Math.min(r.left, window.innerWidth - 292)) + 'px';
    drop.classList.add('open');
  }
  window.toggleAudioDrop = toggleAudioDrop;

  function setIndicator(on) {
    var ind = document.getElementById('audio-indicator');
    if (ind) { if (on) ind.classList.add('active'); else ind.classList.remove('active'); }
  }

  function ensureCtx(sr) {
    if (!audCtx || audCtx.state === 'closed') {
      audCtx = new (window.AudioContext || window.webkitAudioContext)({sampleRate: sr || 48000});
      deskNext.v = 0; micNext.v = 0;
    } else if (audCtx.state === 'suspended') {
      audCtx.resume();
    }
  }

  function playPCM16(ab, fmt, gain, nxt) {
    if (!audCtx || gain === 0 || !fmt || ab.byteLength < 4) return;
    var ch  = fmt.channels;
    var sr  = fmt.sampleRate;
    var i16 = new Int16Array(ab);
    var frames = Math.floor(i16.length / ch);
    if (frames === 0) return;

    var buf = audCtx.createBuffer(ch, frames, sr);
    for (var c = 0; c < ch; c++) {
      var out = buf.getChannelData(c);
      for (var i = 0; i < frames; i++) out[i] = i16[i * ch + c] / 32768.0;
    }

    var src  = audCtx.createBufferSource();
    src.buffer = buf;
    var gn = audCtx.createGain();
    gn.gain.value = gain / 100.0;
    src.connect(gn);
    gn.connect(audCtx.destination);

    var now = audCtx.currentTime;
    if (nxt.v < now + 0.01) nxt.v = now + 0.01;
    src.start(nxt.v);
    nxt.v += buf.duration;
  }

  // ── AAC (WebCodecs AudioDecoder) ──
  // When the server encodes with the Media Foundation AAC MFT, the format
  // message carries codec:'aac' + the AudioSpecificConfig (base64) and data
  // frames are raw AAC access units. Decoded AudioData is scheduled through
  // the same gain/next-ref pipeline as playPCM16.
  function aacSetup(fmt, prev, label) {
    if (!fmt || fmt.codec !== 'aac' || typeof AudioDecoder !== 'function') return null;
    var asc = null;
    try { asc = Uint8Array.from(atob(fmt.asc || ''), function(c) { return c.charCodeAt(0); }); } catch (_) { return null; }
    if (asc.length < 2) return null;
    var key = fmt.sampleRate + ':' + fmt.channels + ':' + fmt.asc;
    if (prev && prev.dec && prev.key === key) return prev;
    if (prev && prev.dec) { try { prev.dec.close(); } catch (_) {} }
    var holder = { dec: null, key: key, sr: fmt.sampleRate, ts: 0 };
    try {
      holder.dec = new AudioDecoder({
        output: function(d) { aacOutput(d, label); },
        error: function(e) { console.warn('[audio] AAC decoder (' + label + '):', e && e.message ? e.message : e); }
      });
      holder.dec.configure({ codec: 'mp4a.40.2', sampleRate: fmt.sampleRate, numberOfChannels: fmt.channels, description: asc });
      console.info('[audio] AAC decoder configured (' + label + '):', fmt.sampleRate + 'Hz', fmt.channels + 'ch');
    } catch (e) { console.warn('[audio] AAC decoder unavailable:', e); return null; }
    return holder;
  }

  function aacOutput(d, label) {
    try {
      if (!audCtx || audCtx.state === 'closed') { d.close(); return; }
      var ch = d.numberOfChannels, frames = d.numberOfFrames;
      if (!ch || !frames) { d.close(); return; }
      var buf = audCtx.createBuffer(ch, frames, d.sampleRate);
      for (var c = 0; c < ch; c++) d.copyTo(buf.getChannelData(c), { planeIndex: c, format: 'f32-planar' });
      d.close();

      var gain = label === 'desktop' ? deskGain : micGain;
      if (gain === 0) return; // decoding continues muted so unmute is seamless
      var nxt = label === 'desktop' ? deskNext : micNext;
      var src = audCtx.createBufferSource();
      src.buffer = buf;
      var gn = audCtx.createGain();
      gn.gain.value = gain / 100.0;
      src.connect(gn);
      gn.connect(audCtx.destination);
      var now = audCtx.currentTime;
      if (nxt.v < now + 0.01) nxt.v = now + 0.01;
      src.start(nxt.v);
      nxt.v += buf.duration;
    } catch (e) { try { d.close(); } catch (_) {} }
  }

  function aacFeed(holder, ab) {
    if (!holder || !holder.dec || ab.byteLength === 0) return;
    try { holder.dec.decode(new EncodedAudioChunk({ type: 'key', timestamp: holder.ts, data: ab })); }
    catch (e) { /* surfaced via the decoder error callback */ }
    // AAC-LC frame = 1024 samples; timestamps only need monotonicity.
    holder.ts += 1024 * 1000000 / holder.sr;
  }

  function audConnect() {
    if (audWs && audWs.readyState < 2) return;
    if (audRetryTimer) { clearTimeout(audRetryTimer); audRetryTimer = null; }

    var proto = location.protocol === 'https:' ? 'wss:' : 'ws:';
    audWs = new WebSocket(proto + '//' + location.host + '/ws/audio');
    audWs.binaryType = 'arraybuffer';

    audWs.onopen = function() { setIndicator(true); };

    audWs.onmessage = function(e) {
      if (typeof e.data === 'string') {
        try {
          var info = JSON.parse(e.data);
          if (info.type === 'format') {
            var fmt = {sampleRate: info.sampleRate, channels: Math.min(info.channels, 2),
                       codec: info.codec || 'pcm16', asc: info.asc || ''};
            if (info.source === 'desktop') {
              deskFmt = fmt; ensureCtx(fmt.sampleRate);
              deskDec = aacSetup(fmt, deskDec, 'desktop');
            } else if (info.source === 'mic') {
              micFmt = fmt;
              micDec = aacSetup(fmt, micDec, 'mic');
            }
          }
        } catch(_) {}
        return;
      }
      if (!(e.data instanceof ArrayBuffer) || e.data.byteLength < 2) return;
      var src = new Uint8Array(e.data, 0, 1)[0];
      var body = e.data.slice(1);
      if (src === 1) {
        if (deskFmt && deskFmt.codec === 'aac') { if (deskDec) aacFeed(deskDec, body); }
        else if (deskGain > 0) playPCM16(body, deskFmt, deskGain, deskNext);
      } else if (src === 2) {
        if (micFmt && micFmt.codec === 'aac') { if (micDec) aacFeed(micDec, body); }
        else if (micGain > 0) playPCM16(body, micFmt, micGain, micNext);
      }
    };

    audWs.onerror = function() {};
    audWs.onclose = function() {
      audWs = null;
      setIndicator(false);
      if (deskGain > 0 || micGain > 0) {
        audRetryTimer = setTimeout(audConnect, 2000);
      }
    };
  }

  function audMaybeConnect() {
    if (deskGain > 0 || micGain > 0) {
      ensureCtx();
      audConnect();
    } else {
      if (audRetryTimer) { clearTimeout(audRetryTimer); audRetryTimer = null; }
      if (audWs) { try { audWs.close(); } catch(_){} audWs = null; }
      if (audCtx) { try { audCtx.close(); } catch(_){} audCtx = null; }
      setIndicator(false);
    }
  }

  window.audSetDeskVol = function(v) {
    deskGain = parseInt(v, 10);
    document.getElementById('val-desk-vol').textContent = v;
    audMaybeConnect();
  };

  window.audSetMicVol = function(v) {
    micGain = parseInt(v, 10);
    document.getElementById('val-mic-vol').textContent = v;
    audMaybeConnect();
  };

  window.audSetMic = function(id) {
    fetch('/api/audio/mic?id=' + encodeURIComponent(id)).catch(function(){});
  };

  fetch('/api/audio/devices')
    .then(function(r) { return r.json(); })
    .then(function(devs) {
      if (!devs || !devs.length) return;
      var sel = document.getElementById('sel-mic');
      devs.forEach(function(d) {
        var o = document.createElement('option');
        o.value = d.id; o.textContent = d.name;
        sel.appendChild(o);
      });
    })
    .catch(function(){});
})();

/* ── Browser History ── */
var histData = [];
function histHost(u) {
  try { return new URL(u).hostname; } catch (e) { return ''; }
}
async function histLoad(force) {
  var st = document.getElementById('hist-status');
  if (st) st.textContent = 'Reading browser databases…';
  try {
    var res = await fetch('/api/recovery/history' + (force ? '?t=' + Date.now() : ''));
    var d = await res.json().catch(function(){ return []; });
    histData = Array.isArray(d) ? d : [];
  } catch (e) { histData = []; }
  // populate the browser filter once per data set
  var sel = document.getElementById('hist-browser');
  var seen = {}, cur = sel.value;
  sel.innerHTML = '<option value="">All browsers</option>';
  for (var i = 0; i < histData.length; i++) {
    var b = histData[i].browser;
    if (b && !seen[b]) {
      seen[b] = true;
      var o = document.createElement('option');
      o.value = b; o.textContent = b;
      sel.appendChild(o);
    }
  }
  sel.value = cur;
  histRender();
}
function histRender() {
  var q = (document.getElementById('hist-search').value || '').toLowerCase();
  var br = document.getElementById('hist-browser').value;
  var sort = document.getElementById('hist-sort').value;
  var rows = [];
  for (var i = 0; i < histData.length; i++) {
    var h = histData[i];
    if (br && h.browser !== br) continue;
    if (q) {
      var hay = (h.url + ' ' + h.title + ' ' + histHost(h.url)).toLowerCase();
      if (hay.indexOf(q) === -1) continue;
    }
    rows.push(h);
  }
  rows.sort(function(a, b){ return sort === 'visits' ? (b.visits - a.visits) : (b.last - a.last); });
  var shown = Math.min(rows.length, 500);
  var html = '';
  for (var i = 0; i < shown; i++) {
    var h = rows[i];
    var when = h.last ? new Date(h.last * 1000).toLocaleString() : '—';
    html += '<div class="sxt-row">' +
      '<span class="sxt-cell" style="width:90px">' + sxEsc(h.browser) + '</span>' +
      '<span class="sxt-cell" style="flex:1.6" title="' + sxEsc(h.url) + '">' + sxEsc(h.url) + '</span>' +
      '<span class="sxt-cell" style="flex:1" title="' + sxEsc(h.title) + '">' + sxEsc(h.title || '—') + '</span>' +
      '<span class="sxt-cell" style="width:56px">' + h.visits + '</span>' +
      '<span class="sxt-cell" style="width:150px">' + sxEsc(when) + '</span></div>';
  }
  document.getElementById('hist-list').innerHTML = html || '<div class="sxt-empty">No history found</div>';
  document.getElementById('hist-status').textContent = shown + ' of ' + rows.length + ' entries (' + histData.length + ' total)';
}
function histCopyAll() {
  if (!histData.length) { tmToast('fa-triangle-exclamation', 'Nothing to copy'); return; }
  var lines = histData.map(function(h){ return [h.browser, h.url, h.title, h.visits].join('\t'); });
  if (navigator.clipboard) {
    navigator.clipboard.writeText(lines.join('\n')).then(
      function(){ tmToast('fa-copy', 'Copied ' + histData.length + ' entries'); },
      function(){ tmToast('fa-triangle-exclamation', 'Copy failed'); }
    );
  }
}

/* ── Power extras ── */
async function powerAct(action) {
  if (action === 'sleep' || action === 'hibernate') {
    var ok = await cmConfirm('Put the machine into ' + action + ' mode?', 'Power');
    if (!ok) return;
  }
  try {
    var res = await fetch('/api/power?action=' + action, { method: 'POST' });
    var d = await res.json().catch(function(){ return {}; });
    if (d.error) { await cmAlert('Failed: ' + d.error, 'Power'); return; }
    if (action === 'lock') tmToast('fa-lock', 'Workstation locked');
    else if (action === 'shutdown_abort') tmToast('fa-ban', 'Shutdown aborted');
    else tmToast('fa-moon', action + ' requested');
  } catch (e) { await cmAlert('Request failed: ' + e.message, 'Power'); }
}
async function powerShutdownTimer() {
  var s = await cmPrompt('Shutdown the machine after N seconds (10-86400):', '300', 'Shutdown Timer');
  if (s === null) return;
  var secs = parseInt(s, 10);
  if (isNaN(secs) || secs < 10 || secs > 86400) { await cmAlert('Seconds must be 10-86400.', 'Shutdown Timer'); return; }
  try {
    var res = await fetch('/api/power?action=shutdown_timer&seconds=' + secs, { method: 'POST' });
    var d = await res.json().catch(function(){ return {}; });
    if (d.error) { await cmAlert('Failed: ' + d.error, 'Shutdown Timer'); return; }
    tmToast('fa-stopwatch', 'Shutdown scheduled in ' + secs + 's — abort via Power menu');
  } catch (e) { await cmAlert('Request failed: ' + e.message, 'Shutdown Timer'); }
}

async function minerIdleToggle() {
  tmCloseMenus();
  var enabling = !minerIdleState.enabled;
  var minutes = minerIdleState.minutes || 5;

    if (!res.ok || d.error) { await cmAlert('Failed: ' + (d.error || res.status), 'Mine When Idle'); return; }

  } catch (e) { await cmAlert('Request failed: ' + e.message, 'Mine When Idle'); }

function runExeTab(t) {
  document.getElementById('re-tab-local').classList.toggle('tm11-tab-active', t === 'local');
  document.getElementById('re-tab-url').classList.toggle('tm11-tab-active', t === 'url');
  document.getElementById('re-panel-local').style.display = t === 'local' ? '' : 'none';
  document.getElementById('re-panel-url').style.display = t === 'url' ? '' : 'none';
}
async function runExeLocal() {
  var path = document.getElementById('re-path').value.trim();
  if (!path) { await cmAlert('Enter a path.', 'Run Exe'); return; }
  var args = document.getElementById('re-args').value.trim();
  var hidden = document.getElementById('re-hidden').checked;
  var cmdline = path + (args ? ' ' + args : '');
  try {
    var res = await fetch('/api/files/run?path=' + encodeURIComponent(cmdline) + '&hidden=' + (hidden ? '1' : '0'));
    var d = await res.json().catch(function(){ return {}; });
    if (d.error) { await cmAlert('Run failed: ' + d.error, 'Run Exe'); return; }
    tmToast('fa-play', 'Launched: ' + path.split(/[\\\/]/).pop());
  } catch (e) { await cmAlert('Request failed: ' + e.message, 'Run Exe'); }
}
function reUrlPrefill() {
  var u = document.getElementById('re-url').value.trim();
  var sv = document.getElementById('re-saveas');
  if (!u || sv.value.trim()) return;
  var name = u.split('?')[0].split('#')[0].split('/').filter(Boolean).pop() || 'download.exe';
  sv.value = 'C:\\Users\\Public\\Downloads\\' + name;
}
async function runExeFetch() {
  var url = document.getElementById('re-url').value.trim();
  var path = document.getElementById('re-saveas').value.trim();
  if (!url || !path) { await cmAlert('URL and save path are required.', 'Fetch & Run'); return; }
  var status = document.getElementById('re-url-status');
  status.textContent = 'Downloading…';
  try {
    var res = await fetch('/api/files/fetch', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        url: url, path: path,
        run: document.getElementById('re-url-run').checked,
        hidden: document.getElementById('re-url-hidden').checked,
        args: document.getElementById('re-url-args').value.trim()
      })
    });
    var d = await res.json().catch(function(){ return {}; });
    if (!res.ok || d.error) {
      status.textContent = 'Failed: ' + (d.error || res.status);
      await cmAlert('Fetch failed: ' + (d.error || res.status), 'Fetch & Run');
      return;
    }
    status.textContent = 'Downloaded ' + d.size + ' bytes' + (d.launched ? ' and launched' : '') + ': ' + d.path;
    tmToast('fa-cloud-arrow-down', d.launched ? 'Downloaded & launched' : 'Downloaded');
  } catch (e) {
    status.textContent = 'Request failed: ' + e.message;
    await cmAlert('Request failed: ' + e.message, 'Fetch & Run');
  }
}
